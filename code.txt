{==================== 四、公共基础 ====================}
N_BAR   := BARSCOUNT(C);
N_SAFE  := MAX(N_BAR, 1);
WARM_OK := N_BAR > WARM_N;

{--- 逐根K线距开盘分钟数(仅分钟线) ---}
BAR_HM  := INTPART(TIME / 100) * 60 + MOD(TIME, 100);
BAR_FO  := IF(IS_MIN_BAR, MAX(IF(BAR_HM <= 690, BAR_HM - 570, BAR_HM - 660), 0), 0);

{--- 盘中成交量预测(仅日线最后一根), 与主图一致 ---}
IN_SESS := FROMOPEN > 0 AND FROMOPEN < TRD_MIN;
IS_PRED := ISLASTBAR AND IN_SESS AND IS_DAY_BAR;
FO_EFF  := MIN(MAX(FROMOPEN, VP_MIN), TRD_MIN);
FRAC_D  := IF(FO_EFF <= 30, 0.20 * FO_EFF / 30,
           IF(FO_EFF <= 60, 0.20 + 0.11 * (FO_EFF - 30) / 30,
           IF(FO_EFF <= 120, 0.31 + 0.17 * (FO_EFF - 60) / 60,
           IF(FO_EFF <= 180, 0.48 + 0.18 * (FO_EFF - 120) / 60,
           IF(FO_EFF <= 210, 0.66 + 0.12 * (FO_EFF - 180) / 30,
                             0.78 + 0.22 * (FO_EFF - 210) / 30)))));
PRED_F  := IF(IS_PRED, IF(CURVE_ON = 1, 1 / FRAC_D, TRD_MIN / FO_EFF), 1);
V_PRED  := IF(USE_AMT = 1, AMOUNT, VOL) * PRED_F;
V_PRED_P1 := IF(N_BAR > 1, REF(V_PRED, 1), V_PRED);
AMT_PRED_RAW := AMOUNT * PRED_F;

PC1     := IF(N_BAR > 1, REF(C, 1), O);
H_P1    := IF(N_BAR > 1, REF(H, 1), H);
L_P1    := IF(N_BAR > 1, REF(L, 1), L);
PREV_C  := MAX(PC1, PRICE_MIN);
PCT_CHG := (C - PREV_C) / PREV_C * 100;
FIRST_C := VALUEWHEN(N_BAR = 1, C);
TR_BASE := MAX(MAX(H - L, ABS(H - PREV_C)), ABS(L - PREV_C));

NET_CHG  := IF(N_BAR > ER_N, ABS(C - REF(C, ER_N)), ABS(C - FIRST_C));
PATH_SUM := IF(N_BAR > ER_N, SUM(ABS(C - PC1), ER_N), SUM(ABS(C - PC1), 0));
ER_SUB   := IF(PATH_SUM < EPS, ER_NEUTRAL, MIN(NET_CHG / (PATH_SUM + EPS), 1));
W_ER     := EMA(ER_SUB, ER_SM);

TIME_DAMP := IF(IS_PRED AND FROMOPEN < DAMP_MIN, MAX(FROMOPEN, DAMP_FLOOR) / DAMP_MIN, 1.0);

K_RNG   := MAX(H - L, MAX(C * RNG_MIN_PCT, ABS(C - PREV_C)));
IN_DIR  := ((C - L) - (H - C)) / (K_RNG + EPS);
RL_DIR  := IF(ABS(IN_DIR) < DIR_FLAT, IF(C > PREV_C, DIR_FB, IF(C < PREV_C, -DIR_FB, 0)), IN_DIR);

ATR_FAST := MA(TR_BASE, ATR_N1);
ATR_MID  := MA(TR_BASE, ATR_N2);
ATR_SLOW := MA(TR_BASE, ATR_N3);
ATR_VAL  := W_ER * ATR_FAST + (1 - W_ER) * (ATR_MIDW * ATR_MID + (1 - ATR_MIDW) * ATR_SLOW);

{MA5/MA45 指数均线}
MA5      := EMA(C, M_PER);
MA45     := EMA(C, M_PER2);
MA5_P1   := IF(N_BAR > 1, REF(MA5, 1), MA5);
MA5_P5   := IF(N_BAR > 5, REF(MA5, 5), MA5);
MA45_P1  := IF(N_BAR > 1, REF(MA45, 1), MA45);

MA_GC    := CROSS(MA5, MA45);

SLP_5      := (MA5 - MA5_P5) / 5 / (MA5 + EPS);
TREND_ST   := SLP_5 > SLP_TH;
STK_OK     := C > MA5 * MA_TOL;
SLF_TR_UP  := MA45 >= MA45_P1 OR C > MA45;

C_BIAS_MA    := (C - MA5) / (MA5 + EPS);
SZ_DOWN_COEF := IF(C_BIAS_MA > BIAS_TH, DN_COEF_UP, DN_COEF_NRM);

TREND_OWN := IF(MA5 > MA45, 35, 0)
         + IF(MA5 > MA5_P1, 25, IF(MA5 >= MA5_P1 * TREND_FLAT, 10, 0))
         + IF(MA45 > MA45_P1, 20, 0)
         + IF(C > MA5, 20, 0);

D_C      := REF(C, D_LAG)#DAY;
D_EMA5   := REF(EMA(C, M_PER), D_LAG)#DAY;
D_EMA45  := REF(EMA(C, M_PER2), D_LAG)#DAY;
D_M5_P   := REF(EMA(C, M_PER), D_LAG + 1)#DAY;
D_M45_P  := REF(EMA(C, M_PER2), D_LAG + 1)#DAY;

TREND_D    := IF(D_EMA5 > D_EMA45, 35, 0)
         + IF(D_EMA5 > D_M5_P, 25, IF(D_EMA5 >= D_M5_P * TREND_FLAT, 10, 0))
         + IF(D_EMA45 > D_M45_P, 20, 0)
         + IF(D_C > D_EMA5, 20, 0);

TREND_DAY     := IF(IS_HTF OR IS_DAY_BAR, TREND_OWN, TREND_D);
MULTI_RESO_OK := TREND_DAY >= TREND_MIN;

HD := H - H_P1;
LD := L_P1 - L;
DMP_RAW := IF(HD > 0 AND HD > LD, HD, 0);
DMM_RAW := IF(LD > 0 AND LD > HD, LD, 0);

TR_F  := EMA(TR_BASE, DI_F);
PDI_F := EMA(DMP_RAW, DI_F) * 100 / (TR_F + EPS);
MDI_F := EMA(DMM_RAW, DI_F) * 100 / (TR_F + EPS);
TR_S  := EMA(TR_BASE, DI_S);
PDI_S := EMA(DMP_RAW, DI_S) * 100 / (TR_S + EPS);
MDI_S := EMA(DMM_RAW, DI_S) * 100 / (TR_S + EPS);

ADX_F   := EMA(ABS(MDI_F - PDI_F) / (MDI_F + PDI_F + EPS) * 100, DI_F);
ADX_S   := EMA(ABS(MDI_S - PDI_S) / (MDI_S + PDI_S + EPS) * 100, DI_S);
ADX_VAL := W_ER * ADX_F + (1 - W_ER) * ADX_S;
ADX_STR := ADX_VAL > ADX_TH;

V_MA      := IF(N_BAR >= V_DYS, MA(V_PRED, V_DYS), SUM(V_PRED, 0) / N_SAFE);
V_MA_SAFE := MAX(V_MA, V_PRED * V_FLOOR_R);
V_SHRK    := V_PRED < V_MA_SAFE * SHR_R;
V_MAS     := IF(N_BAR >= V_SHORT, MA(V_PRED, V_SHORT), SUM(V_PRED, 0) / N_SAFE);
V_MAS_P1  := IF(N_BAR > 1, REF(V_MAS, 1), V_MAS);
VOL_RATIO := V_PRED / (V_MA_SAFE + EPS);

{==================== 五、资金引擎（量比方向） ====================}
SZ_DIR     := IF(C >= PREV_C, VOL_RATIO, -VOL_RATIO * SZ_DOWN_COEF);
M_SZ_OSC_F := EMA(SZ_DIR, SZ_E1) - EMA(SZ_DIR, SZ_E2);
M_SZ_OSC_S := EMA(SZ_DIR, SZ_E2) - EMA(SZ_DIR, SZ_E3);
M_SZ_OSC   := W_ER * M_SZ_OSC_F + (1 - W_ER) * M_SZ_OSC_S;

SZ_STD_RAW := STD(M_SZ_OSC, SZ_STD_N);

SZ_P1      := IF(N_BAR > 1, REF(SZ, 1), SZ);

SZ_SC      := (SZ + SZ_OFF) / SZ_DIV;
SZ_ACC_S   := SZ > SZ_P1 AND SZ > SZ_ACC_MIN;
SZ_VST     := SZ_SC > SZ_VST_TH;
SZ_STR     := SZ_SC > SZ_STR_TH;
SCALPEL    := V_SHRK AND SZ_ACC_S;
EX_CAP     := MIN(VOL_RATIO, EX_RATIO_CAP);

X_PDIR     := IF(C > PREV_C, 1, IF(C < PREV_C, -1, 0));
X_FLOW     := IF(X_PDIR * IN_DIR < 0, FLOW_PEN, 1.0);
X_RDARK    := RL_DIR * (DARK_BASE + (1 - DARK_BASE) * EX_CAP) * X_FLOW;
M_DARK     := IF(W_ER > ER_SWITCH, EMA(X_RDARK, DARK_N_F), EMA(X_RDARK, DARK_N_S));
M_DARK_P1  := IF(N_BAR > 1, REF(M_DARK, 1), M_DARK);

M_D1       := (SZ - SZ_P1) * TIME_DAMP;
M_D1_P     := IF(N_BAR > 1, REF(M_D1, 1), M_D1);
M_D2       := M_D1 - M_D1_P;
M_LEAD     := (M_D1 + M_D2 * LEAD_PHI) * LEAD_K;
M_SZ_MA    := EMA(SZ + M_LEAD * LEAD_GAIN, SM_N);

M_TMC_BASE := W_ER * EMA(SZ, TMC_F) + (1 - W_ER) * EMA(SZ, TMC_S);
M_TMC_P1   := IF(N_BAR > 1, REF(M_TMC_BASE, 1), M_TMC_BASE);
M_TMC_D1   := (M_TMC_BASE - M_TMC_P1) * TIME_DAMP;
M_TMC_D1P  := IF(N_BAR > 1, REF(M_TMC_D1, 1), M_TMC_D1);
M_TMC_D2   := M_TMC_D1 - M_TMC_D1P;
M_TMC_LR   := (M_TMC_D1 + M_TMC_D2 * TMC_PHI) * TMC_K;
M_TMC_LF   := M_TMC_LR * TMC_MIX + EMA(M_TMC_LR, SM_N) * (1 - TMC_MIX);
M_TMC_OBS  := EMA(M_TMC_BASE + M_TMC_LF, SM_N);

M_CROSS_DN  := CROSS(M_TMC_OBS, M_SZ_MA);
M_ACC_BOOST := MAX(0, M_D2);
EXPL_TH_SUB := EXPL_BASE - W_ER * EXPL_ER - MIN(EXPL_ACC_CAP, M_ACC_BOOST * EXPL_ACC);
IS_EXPL_SUB := V_PRED > V_MA_SAFE * EXPL_TH_SUB;
M_CF_TREND  := M_SZ_MA > M_TMC_OBS;

SZ_STD20   := STD(SZ, QUIET_STAT);
SZ_MA20    := MA(SZ, QUIET_STAT);
SZ_WIDTH   := SZ_STD20 * 2 / (ABS(SZ_MA20) + QUIET_EPS);
SQZ_BAR    := SZ_WIDTH < QUIET_TH;
IS_SQUEEZE := N_BAR >= QUIET_STAT AND SQZ_BAR;

M_QUIET    := COUNT((SZ > 0 AND NOT(IS_EXPL_SUB)) OR SQZ_BAR, QUIET_WIN);
M_FLY      := M_CF_TREND AND REF(M_QUIET >= QUIET_MIN, 1) AND SZ > 0 AND IS_EXPL_SUB AND M_DARK > FLY_DARK;

{==================== 五B、副图成交额动量引擎 ====================}
AMT_MA     := IF(N_BAR >= VOL_DAYS, MA(AMT_PRED_RAW, VOL_DAYS), SUM(AMT_PRED_RAW, 0) / N_SAFE);
AMT_MA_PRV := IF(N_BAR > 1, REF(AMT_MA, 1), AMT_MA);
AMT_MA_SAFE := MAX(AMT_MA, MAX(AMT_MA_PRV, AMT_PRED_RAW * AMT_FLOOR_R));

PCT_STD    := IF(N_BAR >= PCT_STD_N, STD(PCT_CHG, PCT_STD_N), 0);
PCT_STD_PRV := IF(N_BAR > 1, REF(PCT_STD, 1), 0);
MAG_BASE   := MAX(MAG_FLOOR, PCT_STD_PRV);

EXPL_TH_BASE := EXPL_TH0 - W_ER * EXPL_ER;
EXPL_RATIO   := MIN(AMT_PRED_RAW / (AMT_MA_SAFE + EPS), EX_RATIO_CAP);
EXPL_GAP     := EXPL_RATIO - EXPL_TH_BASE;
SAFE_EXP_IN  := MAX(-50, MIN(50, -SMOOTH_K * EXPL_GAP));
EXPL_MULT    := EXPL_BASE_MUL + EXPL_RNG_MUL / (1 + EXP(SAFE_EXP_IN));

VOL_DRIVE_CHK := MIN(DRV_HI, MAX(DRV_LO, AMT_PRED_RAW / (AMT_MA_SAFE + EPS)));
CZB_DIR := IF(M_DARK > DARK_TH, 1,
            IF(M_DARK < -DARK_TH, -1,
            IF(C >= PREV_C, 0.5, -0.5)));

CZB := CZB_DIR * (WEIGHT * EXPL_RATIO * EXPL_MULT * VOL_DRIVE_CHK)
     * MAX(MAG_LO, MIN(MAG_HI, ABS(PCT_CHG) / (MAG_BASE + EPS)));

CZB_SM   := EMA(CZB, CZB_N);
SZ_OSC_F := EMA(CZB_SM, OSC_F1) - EMA(CZB_SM, OSC_F2);
SZ_OSC_S := EMA(CZB_SM, OSC_S1) - EMA(CZB_SM, OSC_S2);
SZ_VAL   := EMA(W_ER * SZ_OSC_F + (1 - W_ER) * SZ_OSC_S, OSC_OUT) * SZ_SCALE;

SZ_PRV  := IF(N_BAR > 1, REF(SZ_VAL, 1), SZ_VAL);
D1_RAW  := (SZ_VAL - SZ_PRV) * TIME_DAMP;
D1_PRV  := IF(N_BAR > 1, REF(D1_RAW, 1), D1_RAW);
D2_ACC  := D1_RAW - D1_PRV;
PRV_D2_ACC := IF(N_BAR > 2, REF(D2_ACC, 1), 0);
ACC_BOOST  := MAX(0, PRV_D2_ACC);
EXPL_TH    := EXPL_TH_BASE - MIN(EXPL_ACC_CAP, ACC_BOOST * EXPL_ACC);

LEAD_RAW    := (D1_RAW + D2_ACC * LEAD_D2W) * (LDV_GAIN * LEAD_DAMP);
LEAD_SMOOTH := EMA(LEAD_RAW, LEAD_SM_LEN);
LEAD_FACTOR := LEAD_RAW * LEAD_RAW_W + LEAD_SMOOTH * (1 - LEAD_RAW_W);
SZ_MA_VAL   := EMA(SZ_VAL + LEAD_FACTOR * LEAD_MIX, SZ_MA_LEN);

SZ_GC       := CROSS(SZ_MA_VAL, 0);

{--- 极阴挤压: SQZ_FIX=1 时用近期振幅归一, 避免零轴附近分母趋零失真 ---}
SQZ_STD    := IF(N_BAR >= SQZ_N, STD(SZ_VAL, SQZ_N), 0);
SQZ_MID    := IF(N_BAR >= SQZ_N, MA(SZ_VAL, SQZ_N), SUM(SZ_VAL, 0) / N_SAFE);
SQZ_AMP    := IF(N_BAR >= SQZ_LB, MA(ABS(SZ_VAL), SQZ_LB), SUM(ABS(SZ_VAL), 0) / N_SAFE);
SQZ_WIDTH  := IF(SQZ_FIX = 1, SQZ_STD * 2 / (SQZ_AMP + SQZ_EPS),
                              SQZ_STD * 2 / (ABS(SQZ_MID) + SQZ_EPS));
SQZ_MIN    := IF(N_BAR > SQZ_LB, REF(LLV(SQZ_WIDTH, SQZ_LB), 1), SQZ_WIDTH);

IS_SQZ_V := N_BAR >= MAX(SQZ_WARM, SQZ_LB + 1) AND (SQZ_WIDTH <= SQZ_MIN * SQZ_MULT OR SQZ_WIDTH < SQZ_ABS);
SQZ_FIRE   := IS_SQZ_V AND D1_RAW > 0;

UP_ARR := IF(SZ_VAL > 0, SZ_VAL, 0);
DN_ARR := IF(SZ_VAL < 0, ABS(SZ_VAL), 0);
SUM_UP := IF(N_BAR >= ENERGY_PERIOD, SUM(UP_ARR, ENERGY_PERIOD), SUM(UP_ARR, 0));
SUM_DN := IF(N_BAR >= ENERGY_PERIOD, SUM(DN_ARR, ENERGY_PERIOD), SUM(DN_ARR, 0));
ENERGY_RATIO := IF(SUM_DN < EPS, IF(SUM_UP < EPS, 1, ENERGY_CAP),
                MIN(MAX(SUM_UP / (SUM_DN + EPS), ENERGY_MIN), ENERGY_CAP));

TMC_BASE_F := EMA(SZ_VAL, TMV_F);
TMC_BASE_S := EMA(SZ_VAL, TMV_S);
TMC_BASE   := W_ER * TMC_BASE_F + (1 - W_ER) * TMC_BASE_S;
TMC_PRV    := IF(N_BAR > 1, REF(TMC_BASE, 1), TMC_BASE);
TMC_D1     := (TMC_BASE - TMC_PRV) * TIME_DAMP;
TMC_D1P    := IF(N_BAR > 1, REF(TMC_D1, 1), TMC_D1);
TMC_D2     := TMC_D1 - TMC_D1P;

TMC_LEAD_RAW := (TMC_D1 + TMC_D2 * TMC_D2W) * (TMC_GAIN * TMC_DAMP);
TMC_LEAD_FAC := TMC_LEAD_RAW * TMC_RAW_W
              + EMA(TMC_LEAD_RAW, TMC_LD_SM_LEN) * (1 - TMC_RAW_W);
TMC_OBSERVE  := EMA(TMC_BASE + TMC_LEAD_FAC, TMC_OUT_LEN);
SZ_CROSS_DN  := CROSS(TMC_OBSERVE, SZ_MA_VAL);

IS_EXPL     := AMT_PRED_RAW > AMT_MA_SAFE * EXPL_TH;
CF_TREND_OK := SZ_MA_VAL > TMC_OBSERVE;
Q_ARR       := IF((SZ_VAL > 0 AND NOT(IS_EXPL)) OR IS_SQZ_V, 1, 0);
QUIET_SCORE := IF(N_BAR >= QUIET_WIN, SUM(Q_ARR, QUIET_WIN), SUM(Q_ARR, 0));
QUIET_PRV   := IF(N_BAR > 1, REF(QUIET_SCORE >= QUIET_MIN, 1), 0);

CF_FLY_SIGNAL := CF_TREND_OK AND QUIET_PRV AND SZ_VAL > 0
             AND IS_EXPL AND M_DARK > FLY_DARK;

HAS_FLY     := SUM(CF_FLY_SIGNAL, 0) > 0;
BARS_TO_FLY := IF(HAS_FLY, BARSLAST(CF_FLY_SIGNAL), 0);
VOL_WEAK    := AMT_PRED_RAW < AMT_MA_SAFE * VOLWEAK_R;

{==================== 六、买入信号与衰竭埋伏判定 ====================}
LLV_BASE  := LLV(L, BASE_N);
HH_NEAR   := HHV(H, NEAR_N);
HH_RECENT := HHV(C, RECENT_N);
HHV3      := HHV(H, 3);
LLV3      := LLV(L, 3);          { V2 修复: 原为 HHV(L,3) 笔误 }

BODY_SHRINK := ABS(C - O) / (H - L + EPS) < BODY_SH;
LOW_VOL     := V_PRED < V_MAS * SHR_R OR V_PRED < V_MA_SAFE * LOWV_R;

PLOW_FLAG   := L = LLV(L, DIV_N);
OLD_FLAG    := REF(PLOW_FLAG, DIV_GAP);
PLOW_OLD    := VALUEWHEN(OLD_FLAG, REF(L, DIV_GAP));
SZ_OLD      := VALUEWHEN(OLD_FLAG, REF(SZ, DIV_GAP));
HAS_OLD     := PLOW_OLD > 0;
OLD_AGE     := IF(HAS_OLD, BARSLAST(OLD_FLAG) + DIV_GAP, DIV_GAP);
DIV_BNC_OK  := HHV(C, MIN(MAX(OLD_AGE, 1), DIV_MAXAGE)) - PLOW_OLD > DIV_BOUNCE * ATR_VAL;

SZ_DIVERG   := PLOW_FLAG AND HAS_OLD AND OLD_AGE <= DIV_MAXAGE AND L <= PLOW_OLD
                AND DIV_BNC_OK AND SZ > SZ_OLD AND SZ > REF(SZ, DIV_REC);

SZ_REPAIR   := SZ > SZ_P1 AND M_DARK > M_DARK_P1;

RAW_EXHAUSTION := ((HHV3 - LLV3) / (LLV3 + EPS) * 100 < EXH_RNG OR BODY_SHRINK)
                AND LOW_VOL AND C >= MA5 * CORE_TOL AND C <= MA5 * MA_UP
                AND (C - LLV_BASE) / (LLV_BASE + EPS) * 100 < EXH_BASE
                AND (SLP_5 > -SLP_TH OR C > MA5)
                AND (SZ_DIVERG OR SZ_REPAIR OR SZ > EXH_SZ_MIN);

EXH_CNT        := COUNT(RAW_EXHAUSTION, EXH_WIN);
TRY_EXHAUSTION := RAW_EXHAUSTION AND EXH_CNT = 1;

{--- 多头衰竭（放量滞涨）V2, 与主图一致 ---}
UP_SHD       := H - MAX(C, O);
V_SURGE_TOP  := V_PRED > V_MAS_P1 * TOP_VOL_X;
P_STAGNANT   := PCT_CHG < TOP_CHG AND PCT_CHG > TOP_DROP
                AND UP_SHD > ABS(C - O)
                AND UP_SHD / (H - L + EPS) > TOP_SHADOW;
HI_POS_60    := (C - LLV_BASE) / (HHV(H, BASE_N) - LLV_BASE + EPS);
IN_HIGH_POS  := C > MA(C, 20) AND HI_POS_60 > TOP_POS;
TOP_EXHAUST  := V_SURGE_TOP AND P_STAGNANT AND IN_HIGH_POS;

DONCHIAN_HIGH := REF(HHV(H, N_TRK), 1);
BREAK_POWER   := (C - DONCHIAN_HIGH) / (ATR_VAL + EPS) > BRK_PWR;
RAW_BUY       := CROSS(C, DONCHIAN_HIGH) AND BREAK_POWER AND STK_OK;

KC_UP    := MA5 + KC_MULT * ATR_VAL;
BB_UP    := MA5 + BB_F * STD(C, BB_N);
BB_BREAK := CROSS(C, BB_UP) AND V_PRED > V_PRED_P1 * BB_VOL_R;
GAP_SZ   := (L - H_P1) / (H_P1 + EPS) * 100;
MAIN_GAP := GAP_SZ > GAP_PCT AND C > O AND V_PRED > V_MAS;

BUY_T    := RAW_BUY OR CROSS(C, KC_UP) OR BB_BREAK OR MAIN_GAP;

{--- 开盘屏蔽: 逐根K线时间, 与主图一致 ---}
OPEN_BLK  := IS_MIN_BAR AND BAR_FO > 0 AND BAR_FO <= OPEN_BLOCK;
OPEN_SAFE := NOT(OPEN_BLK);
GATE_OK   := IF(GATE_ON = 1, MULTI_RESO_OK AND OPEN_SAFE, 1);

CAP_FLOW_OK := IF(CAP_EVENT = 1, CROSS(SZ_SC, SZ_SC_MID), SZ_SC > SZ_SC_MID)
            OR MA_GC OR SZ_GC OR (STATE_FLOW = 1 AND SZ_MA_VAL > 0);

CORE_COND := (BUY_T OR CAP_FLOW_OK) AND C > MA5 * CORE_TOL AND SLF_TR_UP;
FLT_PASS  := SCALPEL OR MAIN_GAP OR SZ_STR
         OR (C > HH_NEAR * NEAR_HH AND VOL_RATIO > HH_VOL_R)
         OR (M_DARK > DARK_BUY) OR M_FLY OR MA_GC OR SZ_GC;

TOP_OK    := IF(TOP_BLOCK = 1, NOT(TOP_EXHAUST), 1);
FINAL_BUY := CORE_COND AND FLT_PASS AND GATE_OK AND WARM_OK AND TOP_OK;
SIG_STR   := FINAL_BUY AND (TREND_ST OR SZ_VST OR BREAK_POWER);
SIG_NOR   := FINAL_BUY AND NOT(SIG_STR);

BUY_SIG     := SIG_STR OR SIG_NOR;

{==================== 七、棘轮止损与持仓状态 ====================}
HAS_BUY  := SUM(BUY_SIG, 0) > 0;
B_CNT    := IF(HAS_BUY, BARSLAST(BUY_SIG), NO_BAR);
HOLD_LEN := IF(HAS_BUY, MIN(B_CNT + 1, HOLD_MAX), 1);

BUY_REF_C := VALUEWHEN(BUY_SIG, C);
HH_C_HOLD := HHV(C, HOLD_LEN);
MAX_GAIN  := IF(HAS_BUY, (HH_C_HOLD - BUY_REF_C) / (BUY_REF_C + EPS) * 100, 0);

DYN_X := IF(MAX_GAIN > STRONG_GAIN_PCT OR SZ_SC > SZ_SC_HI, TIGHT_STOP_MULT,
        IF(MAX_GAIN > MID_GAIN_PCT OR SZ_SC > SZ_SC_MID, MID_STOP_MULT,
        IF(TREND_ST AND ADX_STR, ATR_XS, ATR_X)));

TIGHT_LOCK := DYN_X <= TIGHT_STOP_MULT;

RAW_TRAIL    := HH_C_HOLD - DYN_X * ATR_VAL;
STOP_RATCHET := IF(RATCHET_FULL = 1, HHV(RAW_TRAIL, HOLD_LEN),
                IF(HOLD_LEN = 1, RAW_TRAIL, MAX(RAW_TRAIL, REF(RAW_TRAIL, 1))));

CRASH_EX   := C < STOP_RATCHET OR CROSS(MA5, C)
            OR C < MA5 * IF(ADX_STR, CR_ADX, CR_NRM)
            OR B_CNT >= HOLD_MAX;

LAST_CRASH := IF(SUM(CRASH_EX, 0) > 0, BARSLAST(CRASH_EX), NO_BAR);
HOLD_ST    := HAS_BUY AND (LAST_CRASH > B_CNT OR B_CNT = 0);
PREV_HOLD  := IF(N_BAR > 1, REF(HOLD_ST, 1), 0);

{ 与主图 V2 口径一致: 补 WARM_OK, TRY_GATE 控制是否受门控 }
TRY_BUY    := TRY_EXHAUSTION AND NOT(HOLD_ST) AND WARM_OK AND IF(TRY_GATE = 1, GATE_OK, 1);

IS_SUMMER := HOLD_ST AND (MAX_GAIN > STRONG_GAIN_PCT
            OR (C >= HH_RECENT AND SZ_SC < SZ_SC_MID) OR TIGHT_LOCK);
IS_SPRING := HOLD_ST AND NOT(IS_SUMMER);
IS_WINTER := NOT(HOLD_ST) AND WARM_OK AND RAW_EXHAUSTION;
IS_AUTUMN := NOT(HOLD_ST) AND NOT(IS_WINTER);

HOLD_CROSS_DN := HOLD_ST AND IF(DC_SRC = 1, SZ_CROSS_DN, M_CROSS_DN);

{==================== 八、右侧火花确认 ====================}
CONFIRM_BULL := C > O AND C >= PREV_C;
CONFIRM_VOL  := V_PRED > V_MAS * CF_VOL_R;
