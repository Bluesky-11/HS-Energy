{==================== 二、公共基础数据 ====================}
N_BAR   := BARSCOUNT(C);
N_SAFE  := MAX(N_BAR, 1);
WARM_OK := N_BAR > WARM_N;

IN_SESS := FROMOPEN > 0 AND FROMOPEN < TRD_MIN;
IS_PRED := ISLASTBAR AND IN_SESS AND IS_DAY_BAR;
PRED_F  := IF(IS_PRED, TRD_MIN / MAX(FROMOPEN, VP_MIN), 1);

AMT_PRED_RAW := AMOUNT * PRED_F;
V_PRED       := VOL * PRED_F;

PC1     := IF(N_BAR > 1, REF(C, 1), O);
PREV_C  := MAX(PC1, 0.01);
PCT_CHG := (C - PREV_C) / PREV_C * 100;
FIRST_C := VALUEWHEN(N_BAR = 1, C);
TR_BASE := MAX(MAX(H - L, ABS(H - PREV_C)), ABS(L - PREV_C));

NET_CHG  := IF(N_BAR > 10, ABS(C - REF(C, 10)), ABS(C - FIRST_C));
PATH_SUM := IF(N_BAR > 10, SUM(ABS(C - PC1), 10), SUM(ABS(C - PC1), 0));
ER_SUB   := IF(PATH_SUM < EPS, 0.5, MIN(NET_CHG / (PATH_SUM + EPS), 1));
W_ER     := EMA(ER_SUB, 5);

TIME_DAMP := IF(IS_PRED AND FROMOPEN < DAMP_MIN, MAX(FROMOPEN, 5) / DAMP_MIN, 1.0);

K_RNG   := MAX(H - L, MAX(C * 0.002, ABS(C - PREV_C)));
IN_DIR  := ((C - L) - (H - C)) / (K_RNG + EPS);
RL_DIR  := IF(ABS(IN_DIR) < 0.01, IF(C > PREV_C, 0.6, IF(C < PREV_C, -0.6, 0)), IN_DIR);

ATR_FAST := MA(TR_BASE, 12);
ATR_MID  := MA(TR_BASE, 24);
ATR_SLOW := MA(TR_BASE, 40);
ATR_VAL  := W_ER * ATR_FAST + (1 - W_ER) * (0.6 * ATR_MID + 0.4 * ATR_SLOW);

MA13     := EMA(C, M_PER);
SYS_MA55 := EMA(C, M_PER2);
MA13_P1  := IF(N_BAR > 1, REF(MA13, 1), MA13);
MA13_P5  := IF(N_BAR > 5, REF(MA13, 5), MA13);
MA55_P1  := IF(N_BAR > 1, REF(SYS_MA55, 1), SYS_MA55);

SLP_13    := (MA13 - MA13_P5) / 5 / (MA13 + EPS);
TREND_ST  := SLP_13 > SLP_TH;
STK_OK    := C > MA13 * MA13_TOL;
SLF_TR_UP := SYS_MA55 >= MA55_P1 OR C > SYS_MA55;

{多周期趋势门控(与主图一致)}
TREND_OWN := IF(MA13 > SYS_MA55, 35, 0)
           + IF(MA13 > MA13_P1, 25, IF(MA13 >= MA13_P1 * 0.9999, 10, 0))
           + IF(SYS_MA55 > MA55_P1, 20, 0)
           + IF(C > MA13, 20, 0);
D_C      := C#DAY;
D_M13    := EMA(C, M_PER)#DAY;
D_M55    := EMA(C, M_PER2)#DAY;
D_M13_P  := REF(EMA(C, M_PER), 1)#DAY;
D_M55_P  := REF(EMA(C, M_PER2), 1)#DAY;
TREND_D  := IF(D_M13 > D_M55, 35, 0)
          + IF(D_M13 > D_M13_P, 25, IF(D_M13 >= D_M13_P * 0.9999, 10, 0))
          + IF(D_M55 > D_M55_P, 20, 0)
          + IF(D_C > D_M13, 20, 0);
TREND_DAY     := IF(IS_HTF, TREND_OWN, TREND_D);
MULTI_RESO_OK := TREND_DAY >= 55;

HD := H - IF(N_BAR > 1, REF(H, 1), H);
LD := IF(N_BAR > 1, REF(L, 1), L) - L;
DMP_RAW := IF(HD > 0 AND HD > LD, HD, 0);
DMM_RAW := IF(LD > 0 AND LD > HD, LD, 0);
TR_F  := EMA(TR_BASE, 8);
PDI_F := EMA(DMP_RAW, 8) * 100 / (TR_F + EPS);
MDI_F := EMA(DMM_RAW, 8) * 100 / (TR_F + EPS);
TR_S  := EMA(TR_BASE, 20);
PDI_S := EMA(DMP_RAW, 20) * 100 / (TR_S + EPS);
MDI_S := EMA(DMM_RAW, 20) * 100 / (TR_S + EPS);
ADX_F   := EMA(ABS(MDI_F - PDI_F) / (MDI_F + PDI_F + EPS) * 100, 8);
ADX_S   := EMA(ABS(MDI_S - PDI_S) / (MDI_S + PDI_S + EPS) * 100, 20);
ADX_VAL := W_ER * ADX_F + (1 - W_ER) * ADX_S;
ADX_STR := ADX_VAL > 20;

V_MA      := IF(N_BAR >= V_DYS, MA(V_PRED, V_DYS), SUM(V_PRED, 0) / N_SAFE);
V_MA_SAFE := MAX(V_MA, V_PRED * V_FLOOR_R);
V_MA5     := IF(N_BAR >= 5, MA(V_PRED, 5), SUM(V_PRED, 0) / N_SAFE);
VOL_RATIO := V_PRED / (V_MA_SAFE + EPS);

{==================== 三、资金暗流 ====================}
EX_CAP     := MIN(VOL_RATIO, EX_RATIO_CAP);
X_PDIR     := IF(C > PREV_C, 1, IF(C < PREV_C, -1, 0));
X_FLOW     := IF(X_PDIR * IN_DIR < 0, 0.75, 1.0);
X_RDARK    := RL_DIR * (0.60 + 0.40 * EX_CAP) * X_FLOW;
{恒定周期EMA按效率比切换，替代变周期EMA}
M_DARK     := IF(W_ER > 0.6, EMA(X_RDARK, 2), EMA(X_RDARK, 5));

{==================== 四、衰竭与四季判定 ====================}
LLV60 := LLV(L, 60);
HH20  := HHV(H, 20);
HH10C := HHV(C, 10);
HHV3  := HHV(H, 3);
LLV3  := LLV(L, 3);
BODY_SHRINK := ABS(C - O) / (MAX(H - L, C * 0.0001) + EPS) < BODY_SH;
LOW_VOL     := V_PRED < V_MA5 * SHR_R OR V_PRED < V_MA_SAFE * LOWV_R;
RAW_EXHAUSTION := ((HHV3 - LLV3) / (LLV3 + EPS) * 100 < EXH_RNG OR BODY_SHRINK)
                AND LOW_VOL AND C >= MA13 * MA13_TOL AND C <= MA13 * MA13_UP
                AND (C - LLV60) / (LLV60 + EPS) * 100 < EXH_BASE
                AND (SLP_13 > -0.0001 OR C > MA13);
EXH_CNT5       := COUNT(RAW_EXHAUSTION, 5);
TRY_EXHAUSTION := RAW_EXHAUSTION AND EXH_CNT5 = 1;

DONCHIAN_HIGH := REF(HHV(H, N_TRK), 1);
BREAK_POWER   := (C - DONCHIAN_HIGH) / (ATR_VAL + EPS) > BRK_PWR;
RAW_BUY       := CROSS(C, DONCHIAN_HIGH) AND BREAK_POWER AND STK_OK;
KC_UP    := MA13 + KC_MULT * ATR_VAL;
BB_UP    := MA13 + BB_F * STD(C, M_PER);
BB_BREAK := CROSS(C, BB_UP) AND V_PRED > REF(V_PRED, 1) * BB_VOL_R;
H_P1     := IF(N_BAR > 1, REF(H, 1), H);
GAP_SZ   := (L - H_P1) / (H_P1 + EPS) * 100;
MAIN_GAP := GAP_SZ > GAP_PCT AND C > O AND V_PRED > V_MA5;
BUY_T    := RAW_BUY OR CROSS(C, KC_UP) OR BB_BREAK OR MAIN_GAP;

OPEN_BLK  := FROMOPEN > 0 AND FROMOPEN <= OPEN_BLOCK;
OPEN_SAFE := NOT(IF(IS_MIN_BAR, (OPEN_HIST = 1 OR ISLASTBAR) AND OPEN_BLK, IS_PRED AND OPEN_BLK));
GATE_OK   := IF(GATE_ON = 1, MULTI_RESO_OK AND OPEN_SAFE, 1);

CORE_COND := BUY_T AND C > MA13 * CORE_TOL AND SLF_TR_UP;
FLT_PASS  := MAIN_GAP OR (C > HH20 * NEAR_HH AND VOL_RATIO > 1.0) OR (M_DARK > DARK_BUY);

BUY_SIG     := CORE_COND AND FLT_PASS AND WARM_OK AND GATE_OK;
TRY_BUY_RAW := TRY_EXHAUSTION AND NOT(BUY_SIG) AND WARM_OK AND GATE_OK;

HAS_BUY   := SUM(BUY_SIG, 0) > 0;
B_CNT     := IF(HAS_BUY, BARSLAST(BUY_SIG), 9999);
HOLD_LEN  := IF(HAS_BUY, MIN(B_CNT, HOLD_MAX) + 1, 1);
BUY_REF_C := REF(C, HOLD_LEN - 1);
HH_C_HOLD := HHV(C, HOLD_LEN);
MAX_GAIN  := IF(HAS_BUY, (HH_C_HOLD - BUY_REF_C) / (BUY_REF_C + EPS) * 100, 0);
DYN_X := IF(MAX_GAIN > STRONG_GAIN_PCT, TIGHT_STOP_MULT,
        IF(MAX_GAIN > MID_GAIN_PCT, MID_STOP_MULT,
        IF(TREND_ST AND ADX_STR, ATR_XS, ATR_X)));
RAW_TRAIL    := HH_C_HOLD - DYN_X * ATR_VAL;
STOP_RATCHET := HHV(RAW_TRAIL, HOLD_LEN);
CRASH_EX   := C < STOP_RATCHET OR CROSS(MA13, C) OR C < MA13 * IF(ADX_STR, CR_ADX, CR_NRM) OR B_CNT >= HOLD_MAX;
LAST_CRASH := IF(SUM(CRASH_EX, 0) > 0, BARSLAST(CRASH_EX), 9999);
HOLD_ST    := HAS_BUY AND LAST_CRASH > B_CNT;
PREV_HOLD  := IF(N_BAR > 1, REF(HOLD_ST, 1), 0);

{埋伏仅在空仓时有效}
TRY_BUY    := TRY_BUY_RAW AND NOT(HOLD_ST);

IS_SUMMER := HOLD_ST AND (MAX_GAIN > STRONG_GAIN_PCT OR (C >= HH10C) OR DYN_X <= TIGHT_STOP_MULT);
IS_SPRING := HOLD_ST AND NOT(IS_SUMMER);
IS_WINTER := NOT(HOLD_ST) AND WARM_OK AND RAW_EXHAUSTION;
IS_AUTUMN := NOT(HOLD_ST) AND NOT(IS_WINTER);

{==================== 五、副图成交额动量引擎 ====================}
AMT_MA      := IF(N_BAR >= VOL_DAYS, MA(AMT_PRED_RAW, VOL_DAYS), SUM(AMT_PRED_RAW, 0) / N_SAFE);
AMT_MA_PRV  := IF(N_BAR > 1, REF(AMT_MA, 1), AMT_MA);
AMT_MA_SAFE := MAX(AMT_MA, MAX(AMT_MA_PRV, AMT_PRED_RAW * AMT_FLOOR_R));

PCT_MEAN    := SUM(PCT_CHG, 0) / N_SAFE;
PCT_MEAN2   := SUM(PCT_CHG * PCT_CHG, 0) / N_SAFE;
PCT_STD     := IF(N_BAR >= 20, STD(PCT_CHG, 20), SQRT(MAX(PCT_MEAN2 - PCT_MEAN * PCT_MEAN, 0)));
PCT_STD_PRV := IF(N_BAR > 1, REF(PCT_STD, 1), 0);
MAG_BASE    := MAX(MAG_FLOOR, PCT_STD_PRV);

EXPL_TH_BASE := 1.85 - W_ER * 0.60;
EXPL_RATIO   := MIN(AMT_PRED_RAW / (AMT_MA_SAFE + EPS), EX_RATIO_CAP);
EXPL_GAP     := EXPL_RATIO - EXPL_TH_BASE;
SAFE_EXP_IN  := MAX(-50, MIN(50, -SMOOTH_K * EXPL_GAP));
EXPL_MULT    := EXPL_BASE_MUL + EXPL_RNG_MUL / (1 + EXP(SAFE_EXP_IN));
VOL_DRIVE_CHK := MIN(1.5, MAX(0.5, AMT_PRED_RAW / (AMT_MA + EPS)));
CZB_DIR := IF(M_DARK > DARK_TH, 1, IF(M_DARK < -DARK_TH, -1, IF(C >= PREV_C, 0.5, -0.5)));
CZB := CZB_DIR * (WEIGHT * EXPL_RATIO * EXPL_MULT * VOL_DRIVE_CHK)
        * MAX(MAG_LO, MIN(MAG_HI, ABS(PCT_CHG) / (MAG_BASE + EPS)));

CZB_SM5  := EMA(CZB, 5);
SZ_OSC_F := EMA(CZB_SM5, 5) - EMA(CZB_SM5, 13);
SZ_OSC_S := EMA(CZB_SM5, 10) - EMA(CZB_SM5, 26);
SZ_VAL   := EMA(W_ER * SZ_OSC_F + (1 - W_ER) * SZ_OSC_S, 5) * SZ_SCALE;

SZ_PRV  := IF(N_BAR > 1, REF(SZ_VAL, 1), SZ_VAL);
D1_RAW  := (SZ_VAL - SZ_PRV) * TIME_DAMP;
D1_PRV  := IF(N_BAR > 1, REF(D1_RAW, 1), D1_RAW);
D2_ACC  := D1_RAW - D1_PRV;
PRV_D2_ACC := IF(N_BAR > 2, REF(D2_ACC, 1), 0);
ACC_BOOST  := MAX(0, PRV_D2_ACC);
EXPL_TH    := EXPL_TH_BASE - MIN(0.25, ACC_BOOST * 0.5);

LEAD_RAW    := (D1_RAW + D2_ACC * LEAD_D2W) * (LEAD_GAIN * LEAD_DAMP);
LEAD_SMOOTH := EMA(LEAD_RAW, LEAD_SM_LEN);
LEAD_FACTOR := LEAD_RAW * LEAD_ALPHA + LEAD_SMOOTH * (1 - LEAD_ALPHA);
SZ_MA_VAL   := EMA(SZ_VAL + LEAD_FACTOR * LEAD_MIX, SZ_MA_LEN);

SQZ_MEAN  := SUM(SZ_VAL, 0) / N_SAFE;
SQZ_MEAN2 := SUM(SZ_VAL * SZ_VAL, 0) / N_SAFE;
SQZ_STD   := IF(N_BAR >= SQZ_N, STD(SZ_VAL, SQZ_N), SQRT(MAX(SQZ_MEAN2 - SQZ_MEAN * SQZ_MEAN, 0)));
SQZ_MID   := IF(N_BAR >= SQZ_N, MA(SZ_VAL, SQZ_N), SUM(SZ_VAL, 0) / N_SAFE);
SQZ_WIDTH := SQZ_STD * 2 / (ABS(SQZ_MID) + 0.01);
SQZ_MIN   := IF(N_BAR >= SQZ_LB, LLV(SQZ_WIDTH, SQZ_LB), LLV(SQZ_WIDTH, 0));
IS_SQUEEZE := N_BAR >= SQZ_WARM AND (SQZ_WIDTH < SQZ_MIN * SQZ_MULT OR SQZ_WIDTH < SQZ_ABS);
SQZ_FIRE   := IS_SQUEEZE AND D1_RAW > 0;

UP_ARR := IF(SZ_VAL > 0, SZ_VAL, 0);
DN_ARR := IF(SZ_VAL < 0, ABS(SZ_VAL), 0);
SUM_UP := IF(N_BAR >= ENERGY_PERIOD, SUM(UP_ARR, ENERGY_PERIOD), SUM(UP_ARR, 0));
SUM_DN := IF(N_BAR >= ENERGY_PERIOD, SUM(DN_ARR, ENERGY_PERIOD), SUM(DN_ARR, 0));
ENERGY_RATIO := IF(SUM_DN < EPS, 15, MIN(MAX(SUM_UP / (SUM_DN + EPS), 0.1), 15));

TMC_BASE_F := EMA(SZ_VAL, 7);
TMC_BASE_S := EMA(SZ_VAL, 25);
TMC_BASE   := W_ER * TMC_BASE_F + (1 - W_ER) * TMC_BASE_S;
TMC_PRV    := IF(N_BAR > 1, REF(TMC_BASE, 1), TMC_BASE);
TMC_D1     := (TMC_BASE - TMC_PRV) * TIME_DAMP;
TMC_D1P    := IF(N_BAR > 1, REF(TMC_D1, 1), TMC_D1);
TMC_D2     := TMC_D1 - TMC_D1P;
TMC_LEAD_RAW := (TMC_D1 + TMC_D2 * TMC_D2W) * (TMC_GAIN * TMC_DAMP);
TMC_LEAD_FAC := TMC_LEAD_RAW * TMC_ALPHA + EMA(TMC_LEAD_RAW, TMC_LD_SM_LEN) * (1 - TMC_ALPHA);
TMC_OBSERVE  := EMA(TMC_BASE + TMC_LEAD_FAC, TMC_OUT_LEN);
SZ_CROSS_DN := CROSS(TMC_OBSERVE, SZ_MA_VAL);

IS_EXPL     := AMT_PRED_RAW > AMT_MA_SAFE * EXPL_TH;
CF_TREND_OK := SZ_MA_VAL > TMC_OBSERVE;
Q_ARR       := IF((SZ_VAL > 0 AND NOT(IS_EXPL)) OR IS_SQUEEZE, 1, 0);
QUIET_SCORE := IF(N_BAR >= 3, SUM(Q_ARR, 3), SUM(Q_ARR, 0));
{QUIET_PRV 已是前一根的静默状态，CF_FLY 直接使用，避免双重滞后}
QUIET_PRV   := IF(N_BAR > 1, REF(QUIET_SCORE >= 2, 1), 0);
CF_FLY_SIGNAL := CF_TREND_OK AND QUIET_PRV AND SZ_VAL > 0 AND IS_EXPL AND M_DARK > FLY_DARK;

HAS_FLY     := SUM(CF_FLY_SIGNAL, 0) > 0;
BARS_TO_FLY := IF(HAS_FLY, BARSLAST(CF_FLY_SIGNAL), 0);
VOL_WEAK    := AMT_PRED_RAW < AMT_MA_SAFE * VOLWEAK_R;

{==================== 六、进阶：底背离与右侧火花确认 ====================}
{【背离重构】当前为DIV_N周期价格低点且不高于前一低点(间隔>=DIV_GAP)，
  但动能 SZ_VAL 高于前低点时的动能，且短线动能回升 → 真底背离}
PLOW_FLAG   := L = LLV(L, DIV_N);
PLOW_OLD    := VALUEWHEN(REF(PLOW_FLAG, DIV_GAP), REF(L, DIV_GAP));
SZV_OLD     := VALUEWHEN(REF(PLOW_FLAG, DIV_GAP), REF(SZ_VAL, DIV_GAP));
SZ_DIVERG   := PLOW_FLAG AND PLOW_OLD > 0 AND L <= PLOW_OLD AND SZ_VAL > SZV_OLD AND SZ_VAL > REF(SZ_VAL, 2);

CONFIRM_BULL := C > O AND C >= PREV_C;
CONFIRM_VOL  := V_PRED > V_MA5 * CF_VOL_R;
CF_SRC       := IF(CF_SQZ = 1, IS_WINTER OR IS_SQUEEZE, IS_WINTER OR TRY_BUY);
CONFIRM_FLG  := REF(CF_SRC, 1) AND CONFIRM_BULL AND CONFIRM_VOL;

{==================== 七、副图柱体与动量线渲染 ====================}
AUTUMN_UP_BAR := IS_AUTUMN AND C >= O;
AUTUMN_DN_BAR := IS_AUTUMN AND C < O;
STICKLINE(AUTUMN_UP_BAR, 0, SZ_VAL, SEASON_BAR_W, 0), COLORB0B0B0;
STICKLINE(AUTUMN_DN_BAR, 0, SZ_VAL, SEASON_BAR_W, 0), COLOR444444;
STICKLINE(IS_WINTER, 0, SZ_VAL, SEASON_BAR_W, 0), COLORFF00FF;
STICKLINE(IS_SPRING, 0, SZ_VAL, SEASON_BAR_W, 0), COLOR00FFFF;
STICKLINE(IS_SUMMER, 0, SZ_VAL, SEASON_BAR_W, 0), COLOR00AAFF;

STICKLINE(BUY_SIG AND NOT(PREV_HOLD), 0, SZ_VAL, 1.2, 0), COLORCYAN;
STICKLINE(TRY_BUY, 0, SZ_VAL, 1.2, 0), COLOR00FFFF;
