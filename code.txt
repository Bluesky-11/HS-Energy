                  {================ 天道围猎·量能副图  ================}

{━━━━━━━━━━━━━━━━ 一、环境与参数 ━━━━━━━━━━━━━━━━}
IS_INTRA:=PERIOD<3;
IS_INDEX:=CAPITAL=0 OR C=INDEXC;
SKIP_COST:=IS_INDEX OR IS_INTRA;

V_DYS:=IS_INTRA*5+(1-IS_INTRA)*8;
ENERGY_PRD:=IS_INTRA*20+(1-IS_INTRA)*30;
M_PER:=IS_INTRA*5+(1-IS_INTRA)*8;
BB_F:=IS_INTRA*1.5+(1-IS_INTRA)*2.0;
SZ_MA_LEN:=IS_INTRA*12+(1-IS_INTRA)*9;
HOLD_MAX:=IS_INTRA*48+(1-IS_INTRA)*240;
DARK_EMA_W1:=IS_INTRA*6+(1-IS_INTRA)*4;
DARK_EMA_W2:=IS_INTRA*8+(1-IS_INTRA)*3;
CZB_SM_W:=IS_INTRA*8+(1-IS_INTRA)*5;
SZ_EMA_W:=IS_INTRA*5+(1-IS_INTRA)*3;
TMC_BS_W:=IS_INTRA*21+(1-IS_INTRA)*34;
TMC_LD_SM_VAL:=IS_INTRA*8+(1-IS_INTRA)*5;
MA55_PER:=IS_INTRA*13+(1-IS_INTRA)*34;

ATR_X:=1.02;
ATR_XS:=1.42;
ATR_PCT_HI:=0.05;
ATR_PCT_LO:=0.03;
SHR_R:=0.82;
EX_CAP:=5;
EXPL_TH_MIN_AMT:=0.35;
DARK_TH:=0.045;
SZ_SC_MID:=0.90;
WEIGHT:=3.45;
SMOOTH_K:=4.5;
LEAD_GAIN:=2.55;
LEAD_RAW_W:=0.55;
SZ_SC_OFF:=1.9;
SZ_SC_SCL:=0.37037;
TMC_GAIN:=2.35;
TMC_ALPHA:=0.60;
SQZ_FLR_MLT:=1.05;
FLT_MIN_SCR:=3.5;
CHIP_SCR_TH:=11.0;
VWAP_UP_LIM:=1.045;
STP_PRV_MLT:=5;
EPS:=0.0001;
ER_PERIOD:=3;
HOLD_LEN_MIN:=3;
STR_GAIN_PCT:=15;
TGT_STP_MULT:=0.68;
STP_FLOOR_B:=0.75;
STP_FLOOR_C:=0.72;
MA21_STP_FLR:=0.945;
STP_RTCH_BUF:=0.998;
EXIT_COOL_N:=8;

{━━━━━━━━━━━━━━━━ 二、基础数据 ━━━━━━━━━━━━━━━━}
PREV_C:=REF(C,1);
PREV_H:=REF(H,1);
PREV_L:=REF(L,1);
I_C_E:=1/(C+EPS);
RNG_HL:=H-L;
RNG_HL_E:=RNG_HL+EPS;
ABS_CHG:=ABS(C-PREV_C);
TR_BASE:=MAX(MAX(RNG_HL,ABS(H-PREV_C)),ABS(L-PREV_C));
BARS_CNT:=BARSCOUNT(C);
BARS_WARM:=BARS_CNT>ENERGY_PRD;

{--- 窗口极值与均量：统一定义，全文复用 ---}
HH5:=HHV(H,5);
LL5:=LLV(L,5);
HH8:=HHV(H,8);
LL8:=LLV(L,8);
HH10:=HHV(H,10);
LL10:=LLV(L,10);
LL20:=LLV(L,20);
V_MA3:=MA(VOL,3);
V_MA5:=MA(VOL,5);
V_MA8:=MA(VOL,8);

{━━━━━━━━━━━━━━━━ 三、KAUFMAN ER ━━━━━━━━━━━━━━━━}
GAP_COND:=(L>PREV_H) OR (H<PREV_L);

{ 注意：通达信的 IF 两个分支都会求值，COST/WINNER 不会因 SKIP_COST 而被跳过。
  若不需要筹码，可将下面三处替换为：C90_HIGH:=H; C90_LOW:=L; HAS_COST:=0; WIN_RATIO:=50;
  （CHIP_LOCK/CHIP_EFF/CHIP_SAFE 会退化为固定值，速度提升明显） }
C90_HIGH:=IF(SKIP_COST,H,COST(95));
C90_LOW:=IF(SKIP_COST,L,COST(5));
C90_MID:=(C90_HIGH+C90_LOW)*0.5;
C90_SPREAD:=C90_HIGH-C90_LOW;
HAS_COST:=IF(SKIP_COST,0,C90_HIGH>EPS AND C90_MID>EPS AND ABS(C90_SPREAD)/(C90_MID+EPS)>0.0005);

W_ER_RAW:=ABS(C-REF(C,ER_PERIOD))/(MAX(SUM(ABS_CHG,ER_PERIOD),C*0.001)+EPS);
W_ER:=MIN(MAX(W_ER_RAW,GAP_COND*0.90),1);
W_ER_INV:=1-W_ER;
TMC_ER_LEN:=MAX(4,ROUND(8-4*W_ER));

{━━━━━━━━━━━━━━━━ 四、筹码与 VWAP ━━━━━━━━━━━━━━━━}
SCR90:=MIN(MAX(IF(HAS_COST,C90_SPREAD/(C90_MID+EPS)*100,0),0),100);
WIN_RATIO:=IF(SKIP_COST,50,WINNER(C)*100);
CHIP_LOCK:=HAS_COST AND SCR90<CHIP_SCR_TH AND WIN_RATIO>55;
CHIP_EFF:=IF(HAS_COST,1-SCR90*0.0032,0.8);

TOT_AMT:=SUM(AMOUNT,8);
TOT_VOL_RAW:=V_MA8*8;
TOT_VOL_SAFE:=(MAX(TOT_VOL_RAW,C)+EPS)*100;
VWAP_RAW:=TOT_AMT/TOT_VOL_SAFE;
VWAP_VALID:=TOT_VOL_RAW>0 AND VWAP_RAW>C*0.95 AND VWAP_RAW<C*VWAP_UP_LIM;
VWAP_SUPP_VAL:=IF(VWAP_VALID,VWAP_RAW,C);
VWAP_SUPP:=VWAP_VALID AND C>=VWAP_SUPP_VAL*0.98 AND C<=VWAP_SUPP_VAL*1.02;

{━━━━━━━━━━━━━━━━ 五、自适应 ATR 与趋势生命线 ━━━━━━━━━━━━━━━━}
ATR_FAST:=MA(TR_BASE,5);
TR_MA21:=MA(TR_BASE,10);
ATR_SLOW:=MA(TR_BASE,18);
ATR_VAL:=W_ER*ATR_FAST+W_ER_INV*(0.6*TR_MA21+0.4*ATR_SLOW);
I_ATR_VAL_E:=1/(ATR_VAL+EPS);
ATR_PCT:=ATR_VAL*I_C_E;
ATR_ADAPT:=0.90+(ATR_PCT>ATR_PCT_LO)*0.10+(ATR_PCT>ATR_PCT_HI)*0.15;

MA21:=EMA(C,M_PER);
MA21_R1:=REF(MA21,1);
SYS_MA55:=EMA(C,MA55_PER);
MA55_R1:=REF(SYS_MA55,1);
MA55_R2:=REF(MA55_R1,1);

MA21_985:=MA21*0.985;
MA21_992:=MA21*0.992;
MA21_945:=MA21*MA21_STP_FLR;
MA21_018:=MA21*1.018;
I_MA21_E:=1/(MA21+EPS);
SLP_21:=(MA21-MA21_R1)/MAX(M_PER,1)*I_MA21_E;
TREND_ST:=SLP_21>0.00030;
SLF_TR_UP:=MA55_R1>=MA55_R2 OR C>SYS_MA55;
BB_UPPER:=MA21+BB_F*STD(C,M_PER);

{━━━━━━━━━━━━━━━━ 六、DMI 与 ADX ━━━━━━━━━━━━━━━━}
HD:=H-PREV_C;
LD:=PREV_L-L;
DMP_RAW:=(HD>LD AND HD>0)*HD;
DMM_RAW:=(LD>HD AND LD>0)*LD;
TR21_I100:=100/(TR_MA21+EPS);
PDI_S_ADX:=MA(DMP_RAW,10)*TR21_I100;
MDI_S_ADX:=MA(DMM_RAW,10)*TR21_I100;
ADX_VAL:=EMA(ABS(MDI_S_ADX-PDI_S_ADX)/(MDI_S_ADX+PDI_S_ADX+EPS)*100,10);
ADX_STR_S:=ADX_VAL>25;

{━━━━━━━━━━━━━━━━ 七、跨周期趋势 ━━━━━━━━━━━━━━━━}
EMA21_CRS:=IF(IS_INTRA,EMA(C,10),MA21);
EMA55_CRS:=IF(IS_INTRA,EMA(C,34),SYS_MA55);

TD_INTRA_BS:=(MA21>SYS_MA55)*35+(MA21>MA21_R1)*25+(MA21=MA21_R1)*10
            +(SYS_MA55>MA55_R1)*20+(C>MA21)*20;

D_C:=REF(C,1)#DAY;
D_M21:=REF(EMA21_CRS,1)#DAY;
D_M55:=REF(EMA55_CRS,1)#DAY;
D_M21_PR:=REF(EMA21_CRS,2)#DAY;
D_M55_PR:=REF(EMA55_CRS,2)#DAY;

TD_DAILY_BS:=(D_M21>D_M55)*35+(D_M21>D_M21_PR)*25
            +(D_M21>=D_M21_PR*0.9999 AND NOT(D_M21>D_M21_PR))*10
            +(D_M55>D_M55_PR)*20+(D_C>D_M21)*20;

TREND_DAY:=IS_INTRA*TD_INTRA_BS+(1-IS_INTRA)*TD_DAILY_BS;
MULTI_RESO:=TREND_DAY>=55 AND SLF_TR_UP AND SLP_21>-0.001;
TD_STR:=TREND_DAY>=80 AND ADX_STR_S AND TREND_ST AND C>SYS_MA55;

{━━━━━━━━━━━━━━━━ 八、动量量能预测 ━━━━━━━━━━━━━━━━}
SAFE_WIN21:=MIN(V_DYS,MAX(BARS_CNT,3));
CUR_MIN:=MAX(FROMOPEN,1);
IS_LB_IN:=ISLASTBAR AND IS_INTRA;
BAR_MIN_RAW:=MOD(CUR_MIN,10);
BAR_MIN:=BAR_MIN_RAW+(BAR_MIN_RAW=0)*10;
SCALE_SAFE:=MIN(240/MAX(BAR_MIN,5),8);
LB_IN_COND:=IS_LB_IN AND BAR_MIN>=5 AND CUR_MIN<240;

LB_SCALE:=LB_IN_COND*SCALE_SAFE+(1-LB_IN_COND);
AMT_PRD_RW:=AMOUNT*LB_SCALE;
AMT_PRED_RW:=IS_INTRA*EMA(AMT_PRD_RW,5)+(1-IS_INTRA)*AMT_PRD_RW;
AMT_MA21:=MA(AMT_PRD_RW,SAFE_WIN21);
AMT_MA21_SF:=MAX(AMT_MA21,MAX(REF(AMT_MA21,1),AMT_PRED_RW*EXPL_TH_MIN_AMT))+EPS;
I_AMT_MA21_SF:=1/AMT_MA21_SF;

CHG_PCT:=(C-PREV_C)/(PREV_C+EPS)*100;
ABS_CHG_PCT:=ABS(CHG_PCT);
MAG_BASE:=MAX(0.35,REF(STD(CHG_PCT,SAFE_WIN21),1));
EXPL_TH:=1.80-W_ER*0.55;
EXPL_DEN:=EXPL_TH-0.8+EPS;

VOL_PRD_RW:=VOL*LB_SCALE;
VOL_PRED:=IS_INTRA*EMA(VOL_PRD_RW,5)+(1-IS_INTRA)*VOL_PRD_RW;

V_MA:=IF(IS_INTRA,V_MA5,V_MA8);
V_MA_SGE:=MAX(V_MA,VOL*0.15)+EPS;
I_V_MA_SGE:=1/V_MA_SGE;
V_SHRK:=VOL<V_MA_SGE*SHR_R;
EXPL_RATIO:=MIN(AMT_PRED_RW*I_AMT_MA21_SF,EX_CAP);
EXPL_GAP:=EXPL_RATIO-EXPL_TH;
IS_EXPL_SUB:=EXPL_RATIO>EXPL_TH;

EXPL_MULT:=1.28+1.22/(1+EXP(-SMOOTH_K*EXPL_GAP));
MAG_RAT_N:=ABS_CHG_PCT/(MAG_BASE+EPS);
MAG_W_TRD:=MIN(W_ER*1.8,1.0);
VOL_RATIO:=MIN(VOL_PRED*I_V_MA_SGE,EX_CAP);

VOL_LEAD_RAW:=EMA(VOL_RATIO/(MAG_RAT_N+1.0),5)*0.35;
VOL_LEAD_N:=MIN(MAX(VOL_LEAD_RAW,0.2),1.8);
MAG_CLAMP:=MAX(0.5,MIN(2.0,MAG_W_TRD*MAX(0.5,MIN(2.0,MAG_RAT_N))+(1-MAG_W_TRD)*VOL_LEAD_N));

{━━━━━━━━━━━━━━━━ 九、暗流引擎（含换手率重塑） ━━━━━━━━━━━━━━━━}
IS_LIMIT_UP:=H=L AND C>PREV_C;
IS_LIMIT_DN:=H=L AND C<PREV_C;
INTRA_DIRB:=IF(RNG_HL>EPS,((C-L)-(H-C))/RNG_HL,0);
INTRA_DIRR:=IS_LIMIT_UP-IS_LIMIT_DN+INTRA_DIRB;

TURNOVER_VAL:=IF(SKIP_COST,1.0,VOL/(CAPITAL+EPS)*100);
TURNOVER_MA8:=MA(TURNOVER_VAL,8);
RT_FACTOR:=MIN(MAX(TURNOVER_VAL/(TURNOVER_MA8+EPS),0.25),2.8);
TE_EFF:=CHG_PCT/(TURNOVER_VAL+0.1);
TE_FACTOR:=MIN(MAX(1.0+TE_EFF*0.08,0.55),1.9);

DARK_FLW_R:=INTRA_DIRR*0.40*(0.52+0.28*EXPL_RATIO+0.20*RT_FACTOR)*CHIP_EFF*TE_FACTOR;
DARK_FLW_R1:=REF(DARK_FLW_R,1);

DARK_FLOW:=EMA(DARK_FLW_R,DARK_EMA_W1);
DARK_ACCEL:=EMA((DARK_FLW_R-DARK_FLW_R1)*1.20,DARK_EMA_W2);

DF_REF1:=REF(DARK_FLOW,1);
DF_OK_SOFT:=DARK_FLOW>-0.02;

DF_POS:=DARK_FLOW>DARK_TH;
DF_NEG:=DARK_FLOW<-DARK_TH;
DF_MID:=1-DF_POS-DF_NEG;
DF_MID_SGN:=DF_MID*((C>=PREV_C)-0.5);
CZB_DIR:=DF_POS-DF_NEG+DF_MID_SGN;

CZB_CORE:=WEIGHT*EXPL_RATIO*EXPL_MULT;
CZB:=CZB_DIR*CZB_CORE*MAG_CLAMP;
CZB_SM5:=EMA(CZB,CZB_SM_W);

CZB_EMA5:=EMA(CZB_SM5,5);
CZB_EMA8:=EMA(CZB_SM5,8);
CZB_EMA10:=EMA(CZB_SM5,10);
CZB_EMA20:=EMA(CZB_SM5,20);
SZ_RAW:=EMA(W_ER*(CZB_EMA5-CZB_EMA10)+W_ER_INV*(CZB_EMA8-CZB_EMA20),SZ_EMA_W)*0.75*CHIP_EFF;
SZ:=MAX(MIN(SZ_RAW+DARK_ACCEL*0.65,15),-15);
SZ_REF1:=REF(SZ,1);
D1_RAW:=SZ-SZ_REF1;
MOM_BOOST:=1.0+(CHG_PCT>0)*CHG_PCT*0.10*W_ER*1.2;

{ 变周期平滑统一用 DMA(ALPHA=2/(N+1))，结果确定且不回溯 }
SZ_ADJ_LEN:=MAX(IS_INTRA*8+(1-IS_INTRA)*5,INTPART(SZ_MA_LEN*(1-W_ER)+6*W_ER+0.5));
SZ_ADJ_ALPHA:=2/(SZ_ADJ_LEN+1);
LEAD_RAW_A:=MIN(MAX((D1_RAW+DARK_ACCEL*0.35)*LEAD_GAIN*MOM_BOOST,-8.5),8.5);
LEAD_SMOOTH:=DMA(LEAD_RAW_A,SZ_ADJ_ALPHA);
LEAD_FACTOR:=LEAD_RAW_A*LEAD_RAW_W+LEAD_SMOOTH*(1-LEAD_RAW_W);
SZ_MA_VAL:=DMA(SZ+LEAD_FACTOR*0.8,SZ_ADJ_ALPHA);
SZ_MA_R1:=REF(SZ_MA_VAL,1);
SZ_MA_UP:=SZ_MA_VAL>SZ_MA_R1;

SZ_P_UP:=SZ_MA_UP AND DARK_FLOW>0.008 AND W_ER>0.32;
SZ_HIGH_PUR:=SZ_P_UP AND SZ_MA_VAL>0.25;

SZ_MA21:=MA(SZ,10);
SQZ_WIDTH:=STD(SZ,8)*2/(ABS(SZ_MA21)+0.01);
IS_SQUEEZE:=SQZ_WIDTH<LLV(SQZ_WIDTH,20)*SQZ_FLR_MLT;

DIR_VOL_RW:=INTRA_DIRR*VOL;
DIR_VOL_AC:=EMA(DIR_VOL_RW*I_V_MA_SGE,8);
DIR_VOL_NM:=MIN(MAX(EMA(DIR_VOL_AC,5),-2.5),2.5);
DIR_VOL_ABS:=ABS(DIR_VOL_RW);

VOL_SUM_EP:=SUM(VOL,10)+EPS;
VOL_ABS_SUM10:=SUM(DIR_VOL_ABS,10);
VOL_ACCUM:=SUM(DIR_VOL_RW,10);
VOL_ACC_POS:=(VOL_ABS_SUM10+VOL_ACCUM)*0.5;
VOL_ACC_EP:=VOL_ABS_SUM10+EPS*10;
ENERGY_CHG:=EMA(VOL_ACCUM/VOL_ACC_EP,5);
DIR_CONSIST:=VOL_ACC_POS/VOL_SUM_EP>0.55;

{━━━━━━━━━━━━━━━━ 十、评分与 TMC ━━━━━━━━━━━━━━━━}
W_S1:=0.20+W_ER*0.20;
W_S2:=0.35-W_ER*0.15;
W_S3:=0.30-W_ER*0.10;
W_S4:=0.15+W_ER*0.15;
DF_NORM:=MIN(MAX((DARK_FLOW+0.05)*10,0),1.0);

ESC_P1:=MIN(MAX((EXPL_RATIO-1.0)/EXPL_DEN,0),1.0)*W_S1;
ESC_P2:=MAX(ENERGY_CHG,0)*IF(DIR_CONSIST,1.0,0.6)*W_S2;
ESC_P3:=MIN(VOL_RATIO,3.0)*0.33333*(1-V_SHRK*0.2)*W_S3;
ESC_P4:=DF_NORM*CHIP_EFF*W_S4;
ENERGY_SCR:=MIN(ROUND((ESC_P1+ESC_P2+ESC_P3+ESC_P4)*100),100);
ENG_VAL_OK:=ENERGY_SCR>=66;
ENG_EXPLODE:=ENERGY_SCR>=82;
HI_RES_ENG:=ENERGY_SCR>=80;

TMC_OUT_LN:=MAX(IS_INTRA*10+(1-IS_INTRA)*8,INTPART(14.5*(1-W_ER*0.50)+0.5));
TMC_ER_ALPHA:=2/(TMC_ER_LEN+1);
TMC_OUT_ALPHA:=2/(TMC_OUT_LN+1);
TMC_BASE_F:=DMA(SZ,TMC_ER_ALPHA);
TMC_BASE_S:=EMA(SZ,TMC_BS_W);
TMC_BASE:=W_ER*TMC_BASE_F+W_ER_INV*TMC_BASE_S;
TMC_BS_LED:=TMC_BASE+DIR_VOL_NM*0.18*W_ER_INV;
TMC_D1:=TMC_BS_LED-REF(TMC_BS_LED,1);
TMC_D1_R1:=REF(TMC_D1,1);
TMC_LD_RAW:=MIN(MAX((TMC_D1+(TMC_D1-TMC_D1_R1)*0.55)*TMC_GAIN,-5.8),5.8);
TMC_LD_EMA3:=EMA(TMC_LD_RAW,TMC_LD_SM_VAL);
TMC_LD_FAC:=TMC_LD_RAW*(TMC_ALPHA+W_ER*0.40)+TMC_LD_EMA3*(W_ER_INV*0.40);
TMC_OBSERVE:=DMA(TMC_BS_LED+TMC_LD_FAC,TMC_OUT_ALPHA);
TMC_SLP_UP:=TMC_OBSERVE>REF(TMC_OBSERVE,1);
SZ_POS:=SZ>0;

QUIET_COND:=(SZ_POS AND NOT(IS_EXPL_SUB)) OR IS_SQUEEZE;
QUIET_SCR:=COUNT(QUIET_COND,5);
CF_FLY_SIG:=SZ_MA_VAL>TMC_OBSERVE AND QUIET_SCR>=2 AND SZ_POS AND IS_EXPL_SUB AND DF_OK_SOFT AND TMC_SLP_UP AND SZ_HIGH_PUR;

DF_CROSS_DN:=CROSS(TMC_OBSERVE,SZ_MA_VAL) AND HHV(ADX_VAL,10)-ADX_VAL>3;
SZ_SC:=MIN(MAX((SZ+SZ_SC_OFF)*SZ_SC_SCL,0),2.0);
SZ_ACC_S:=D1_RAW>0 AND SZ>-0.85;
SZ_VST:=SZ_SC>1.15;
SZ_STR:=SZ_SC>0.72;
SCALPEL:=V_SHRK AND SZ>-1.0 AND SZ_ACC_S;

{━━━━━━━━━━━━━━━━ 十一、VCP ━━━━━━━━━━━━━━━━}
AMP_T1:=(HH10-LL10)/(LL10+EPS)*100;
AMP_T2:=(HH8-LL8)/(LL8+EPS)*100;
AMP_T3:=(HH5-LL5)/(LL5+EPS)*100;

IS_VCP_DEC:=AMP_T1>AMP_T2 AND AMP_T2>AMP_T3
            AND AMP_T3<9.5*ATR_ADAPT
            AND AMP_T1<MAX(25.0*ATR_ADAPT,22.0);

VOL_PEAK_T2:=HHV(VOL,8);
IS_VOL_STEP_DRY:=HHV(VOL,5)<VOL_PEAK_T2 AND VOL_PEAK_T2<HHV(VOL,10);

V_MA5_R3:=REF(V_MA5,3);
IS_VOL_DEC:=(V_MA8>V_MA5_R3 AND V_MA5_R3>V_MA3) OR IS_VOL_STEP_DRY;
IS_VOL_DRY:=V_MA3*I_V_MA_SGE<0.58 OR VOL<V_MA_SGE*0.50;

PIVOT_DIST:=(HH10-C)*I_C_E*100;
IS_PVT_NEAR:=PIVOT_DIST<3.5 AND PIVOT_DIST>=-0.5;
IS_PVT_ZONE:=C>=HH5*0.985 AND IS_PVT_NEAR;
BOTTOM_RSG:=LL5>LL8*0.995 AND LL8>LL10*0.99;

VCP_FLOW_OK:=DF_OK_SOFT AND DARK_FLOW>=DF_REF1;

VCP_S1:=IS_VCP_DEC*35+(1-IS_VCP_DEC)*(AMP_T2>AMP_T3)*15;
VCP_S2:=(IS_VOL_DEC AND IS_VOL_DRY)*30+IS_VOL_DRY*(1-(IS_VOL_DEC AND IS_VOL_DRY))*20;
VCP_S3:=IS_PVT_ZONE*20+(1-IS_PVT_ZONE)*(IS_PVT_NEAR*10+(1-IS_PVT_NEAR)*5);
VCP_S4:=VCP_FLOW_OK*(IF(SZ_SC>0.85 OR CHIP_LOCK,15,8));
VCP_SCORE:=MIN(VCP_S1+VCP_S2+VCP_S3+VCP_S4,100);

VCP_OPTIMAL:=VCP_SCORE>=72 AND BOTTOM_RSG AND IS_VOL_DRY;

{━━━━━━━━━━━━━━━━ 十二、突破与买点 ━━━━━━━━━━━━━━━━}
BIAS21:=(C-MA21)*I_MA21_E*100;

DIST_LL20:=(C-LL20)/(LL20+EPS)*100;
BODY_LEN:=ABS(C-O);
BODY_SHRK:=BODY_LEN/RNG_HL_E<0.35;
LOW_VOL:=VOL<V_MA5*0.75 AND VOL<V_MA_SGE*0.70;
FLOW_OK:=DARK_FLOW>=-0.015;

CHIP_SAFE:=NOT(HAS_COST) OR (C>=C90_MID AND SCR90<CHIP_SCR_TH);

RAW_EXHAUST:=C>LL20*1.03 AND C>SYS_MA55 AND C>=MA21_992 AND C<=MA21_018
