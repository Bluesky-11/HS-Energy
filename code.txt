{==================================================================
   天道围猎·资金气数动量图 V3 (统一逻辑版, 副图)
   - 四季状态(春夏秋冬)/战术节点/止损/持仓 全部来自主图 SCALPEL AMBUSH 的同一套逻辑
   - 副图柱体高度与曲线仍使用本图自己的"成交额动量引擎"(SZ_VAL / SZ_MA / TMC)
   - 命名约定: 主图逻辑中与副图引擎重名的变量一律加 M_ 前缀
        M_DARK=主图暗流  M_SZ_MA/M_TMC_OBS=主图超前线  M_CROSS_DN=主图死叉  M_FLY=主图起爆
   - 主图公式里的买入条件、止损棘轮、四季判定保持原样, 不要在这里单独改动阈值
==================================================================}

{==================== 一、参数区 ====================}
{--- 主图逻辑参数 ---}
N_TRK      := 15;     {突破观察周期}
ATR_X      := 1.06;   {普通趋势ATR止损}
ATR_XS     := 1.48;   {强趋势ATR止损}
V_DYS      := 15;     {量能均线周期}
SHR_R      := 0.85;   {缩量阈值}
BB_F       := 2.0;    {布林倍数}
HOLD_MAX   := 240;    {最长持仓周期}
M_PER      := 25;     {生命线周期 (MA25)}
EPS        := 0.0001;
MA25_TOL   := 0.985;
CORE_TOL   := 0.982;
STRONG_GAIN_PCT := 20;
TIGHT_STOP_MULT := 0.68;
MID_STOP_MULT   := 0.88;
MID_GAIN_PCT    := 12;
SZ_SC_MID  := 0.90;
SZ_SC_HI   := 1.30;
SZ_OFF     := 1.9;
SZ_DIV     := 2.7;
SZ_STD_N   := 80;
WARM_N     := 60;
OPEN_BLOCK := 40;     {开盘后禁止出信号的分钟数}

{--- 时间/会话 (通达信 PERIOD: 0=5分 1=15分 2=30分 3=60分 4=日 5=周 6=月 7=1分 8=多分钟) ---}
TRD_MIN    := 240;
VP_MIN     := 30;
DAY_PRD    := 4;      {日线对应的PERIOD值, 请与你的软件核对}
DAMP_MIN   := 60;

{--- 副图成交额动量引擎参数 ---}
VOL_DAYS      := 10;
WEIGHT        := 3.55;
SMOOTH_K      := 6.8;
ENERGY_PERIOD := 66;
EX_RATIO_CAP  := 8;
DARK_TH       := 0.05;
SEASON_BAR_W  := 0.35;

LEAD_GAIN     := 2.95;
LEAD_DAMP     := 0.9;
LEAD_D2W      := 0.5;
LEAD_ALPHA    := 0.75;
LEAD_SM_LEN   := 2;
LEAD_MIX      := 1.15;
SZ_MA_LEN     := 3;
SZ_SCALE      := 0.69;

TMC_GAIN      := 2.35;
TMC_DAMP      := 0.85;
TMC_D2W       := 0.4;
TMC_ALPHA     := 0.75;
TMC_LD_SM_LEN := 3;
TMC_OUT_LEN   := 3;

MAG_FLOOR     := 0.25;
MAG_LO        := 0.5;
MAG_HI        := 2.0;

DN_COEF_UP    := 0.65;
DN_COEF_NRM   := 0.85;
BIAS_TH       := 0.01;
EXPL_BASE_MUL := 1.28;
EXPL_RNG_MUL  := 1.22;

SQZ_WARM      := 20;
SQZ_N         := 20;
SQZ_LB        := 40;
SQZ_MULT      := 1.25;
SQZ_ABS       := 0.22;

{==================== 二、公共基础 (主/副共用) ====================}
N_BAR   := BARSCOUNT(C);
WARM_OK := N_BAR > WARM_N;

IS_DAY_BAR := PERIOD = DAY_PRD;
IS_MIN_BAR := PERIOD <= 3 OR PERIOD = 7 OR PERIOD = 8;
IS_HTF     := NOT(IS_DAY_BAR OR IS_MIN_BAR);
IN_SESS    := FROMOPEN > 0 AND FROMOPEN < TRD_MIN;
IS_PRED    := ISLASTBAR AND IN_SESS AND IS_DAY_BAR;             {实时日线最后一根K}
IS_LIVE_ANY:= ISLASTBAR AND IN_SESS AND (IS_DAY_BAR OR IS_MIN_BAR);
PRED_F     := IF(IS_PRED, TRD_MIN / MAX(FROMOPEN, VP_MIN), 1);

AMT_PRED_RAW := AMOUNT * PRED_F;
V_PRED       := VOL * PRED_F;

PC1     := IF(N_BAR > 1, REF(C, 1), O);
PREV_C  := MAX(PC1, 0.01);
PCT_CHG := (C - PREV_C) / PREV_C * 100;
FIRST_C := VALUEWHEN(N_BAR = 1, C);
TR_BASE := MAX(MAX(H - L, ABS(H - PREV_C)), ABS(L - PREV_C));

NET_CHG  := IF(N_BAR > 10, ABS(C - REF(C, 10)), ABS(C - FIRST_C));
PATH_SUM := IF(N_BAR >= 10, SUM(ABS(C - PC1), 10), SUM(ABS(C - PC1), 0));
ER_SUB   := IF(PATH_SUM < EPS, 0.5, MIN(NET_CHG / (PATH_SUM + EPS), 1));
W_ER     := EMA(ER_SUB, 5);

{ 开盘阻尼: 仅实时日线最后一根K }
TIME_DAMP := IF(IS_PRED AND FROMOPEN < DAMP_MIN, MAX(FROMOPEN, 5) / DAMP_MIN, 1.0);

{ K线内部方向 (主/副共用) }
K_RNG   := MAX(H - L, MAX(C * 0.002, ABS(C - PREV_C)));
IN_DIR  := ((C - L) - (H - C)) / K_RNG;
RL_DIR  := IF(ABS(IN_DIR) < 0.01, IF(C > PREV_C, 0.6, IF(C < PREV_C, -0.6, 0)), IN_DIR);

{ 均线与自适应ATR }
ATR_FAST := MA(TR_BASE, 12);
ATR_MID  := MA(TR_BASE, 24);
ATR_SLOW := MA(TR_BASE, 40);
ATR_VAL  := W_ER * ATR_FAST + (1 - W_ER) * (0.6 * ATR_MID + 0.4 * ATR_SLOW);

MA25     := EMA(C, M_PER);
SYS_MA55 := EMA(C, 55);
MA25_P1  := IF(N_BAR > 1, REF(MA25, 1), MA25);
MA25_P5  := IF(N_BAR > 5, REF(MA25, 5), MA25);
MA55_P1  := IF(N_BAR > 1, REF(SYS_MA55, 1), SYS_MA55);

SLP_25    := (MA25 - MA25_P5) / 5 / (MA25 + EPS);
TREND_ST  := SLP_25 > 0.00025;
STK_OK    := C > MA25 * MA25_TOL;
SLF_TR_UP := SYS_MA55 >= MA55_P1 OR C > SYS_MA55;

C_BIAS_MA    := (C - MA25) / (MA25 + EPS);
SZ_DOWN_COEF := IF(C_BIAS_MA > BIAS_TH, DN_COEF_UP, DN_COEF_NRM);

{ 多周期趋势分 }
TREND_OWN := IF(MA25 > SYS_MA55, 35, 0)
           + IF(MA25 > MA25_P1, 25, IF(MA25 >= MA25_P1 * 0.9999, 10, 0))
           + IF(SYS_MA55 > MA55_P1, 20, 0)
           + IF(C > MA25, 20, 0);
D_C      := C#DAY;
D_M25    := EMA(C, 25)#DAY;
D_M55    := EMA(C, 55)#DAY;
D_M25_P  := REF(EMA(C, 25), 1)#DAY;
D_M55_P  := REF(EMA(C, 55), 1)#DAY;
TREND_D  := IF(D_M25 > D_M55, 35, 0)
          + IF(D_M25 > D_M25_P, 25, IF(D_M25 >= D_M25_P * 0.9999, 10, 0))
          + IF(D_M55 > D_M55_P, 20, 0)
          + IF(D_C > D_M25, 20, 0);
TREND_DAY     := IF(IS_HTF, TREND_OWN, TREND_D);
MULTI_RESO_OK := TREND_DAY >= 55;

{ DMI / ADX }
HD := H - IF(N_BAR > 1, REF(H, 1), H);
LD := IF(N_BAR > 1, REF(L, 1), L) - L;
DMP_RAW := IF(HD > 0 AND HD > LD, HD, 0);
DMM_RAW := IF(LD > 0 AND LD > HD, LD, 0);

TR_F  := EMA(TR_BASE, 8);
PDI_F := EMA(DMP_RAW, 8) * 100 / (TR_F + EPS);
MDI_F := EMA(DMM_RAW, 8) * 100 / (TR_F + EPS);
TR_S  := EMA(TR_BASE, 20);
PDI_S := EMA(DMP_RAW, 20) * 100 / (TR_S + EPS);
MDI_S := EMA(DMM_RAW, 20) * 100 / (TR_S + EPS);

ADX_F   := EMA(ABS(MDI_F - PDI_F) / (MDI_F + PDI_F + EPS) * 100, 8);
ADX_S   := EMA(ABS(MDI_S - PDI_S) / (MDI_S + PDI_S + EPS) * 100, 20);
ADX_VAL := W_ER * ADX_F + (1 - W_ER) * ADX_S;
ADX_STR := ADX_VAL > 20;

{ 量能 (成交量口径) }
V_MA      := IF(N_BAR >= V_DYS, MA(V_PRED, V_DYS), SUM(V_PRED, 0) / N_BAR);
V_MA_SAFE := MAX(V_MA, V_PRED * 0.15);
V_SHRK    := V_PRED < V_MA_SAFE * SHR_R;
V_MA5     := IF(N_BAR >= 5, MA(V_PRED, 5), SUM(V_PRED, 0) / N_BAR);
VOL_RATIO := V_PRED / (V_MA_SAFE + EPS);

{==================== 三、主图资金引擎 (SZ 体系, 加 M_ 前缀) ====================}
SZ_DIR    := IF(C >= PREV_C, VOL_RATIO, -VOL_RATIO * SZ_DOWN_COEF);

M_SZ_OSC_F := EMA(SZ_DIR, 5) - EMA(SZ_DIR, 15);
M_SZ_OSC_S := EMA(SZ_DIR, 15) - EMA(SZ_DIR, 45);
M_SZ_OSC   := W_ER * M_SZ_OSC_F + (1 - W_ER) * M_SZ_OSC_S;

SZ_STD_RAW := STD(M_SZ_OSC, MAX(MIN(SZ_STD_N, N_BAR), 3));
SZ_STD     := MAX(IF(SZ_STD_RAW >= 0, SZ_STD_RAW, 0.05), 0.015);
SZ_RAW     := M_SZ_OSC / SZ_STD;
SZ         := IF(SZ_RAW >= 0 OR SZ_RAW < 0, SZ_RAW, 0);
SZ_P1      := IF(N_BAR > 1, REF(SZ, 1), SZ);

SZ_SC      := (SZ + SZ_OFF) / SZ_DIV;
SZ_ACC_S   := SZ > SZ_P1 AND SZ > -0.85;
SZ_VST     := SZ_SC > 1.15;
SZ_STR     := SZ_SC > 0.72;
SCALPEL    := V_SHRK AND SZ_ACC_S;

EX_CAP     := MIN(VOL_RATIO, EX_RATIO_CAP);
M_DARK     := EMA(RL_DIR * (0.65 + 0.35 * EX_CAP), 3);

M_D1       := (SZ - SZ_P1) * TIME_DAMP;
M_D1_P     := IF(N_BAR > 1, REF(M_D1, 1), M_D1);
M_D2       := M_D1 - M_D1_P;
M_LEAD_RAW := (M_D1 + M_D2 * 0.618) * 2.95;
M_LEAD     := M_LEAD_RAW * 0.85 + EMA(M_LEAD_RAW, 1) * 0.15;
M_SZ_MA    := EMA(SZ + M_LEAD * 1.25, 2);

M_TMC_BASE := W_ER * EMA(SZ, 7) + (1 - W_ER) * EMA(SZ, 25);
M_TMC_P1   := IF(N_BAR > 1, REF(M_TMC_BASE, 1), M_TMC_BASE);
M_TMC_D1   := (M_TMC_BASE - M_TMC_P1) * TIME_DAMP;
M_TMC_D1P  := IF(N_BAR > 1, REF(M_TMC_D1, 1), M_TMC_D1);
M_TMC_D2   := M_TMC_D1 - M_TMC_D1P;
M_TMC_LR   := (M_TMC_D1 + M_TMC_D2 * 0.5) * 2.35;
M_TMC_LF   := M_TMC_LR * 0.82 + EMA(M_TMC_LR, 2) * 0.18;
M_TMC_OBS  := EMA(M_TMC_BASE + M_TMC_LF, 2);

M_CROSS_DN := CROSS(M_TMC_OBS, M_SZ_MA);

M_ACC_BOOST := MAX(0, M_D2);
EXPL_TH_SUB := 1.85 - W_ER * 0.60 - MIN(0.25, M_ACC_BOOST * 0.5);
IS_EXPL_SUB := V_PRED > V_MA_SAFE * EXPL_TH_SUB;
M_CF_TREND  := M_SZ_MA > M_TMC_OBS;
SZ_STD20    := STD(SZ, 20);
SZ_MA20     := MA(SZ, 20);
M_QUIET     := COUNT((SZ > 0 AND NOT(IS_EXPL_SUB)) OR (SZ_STD20 * 2 / (ABS(SZ_MA20) + 0.01) < 0.22), 3);
M_FLY       := M_CF_TREND AND REF(M_QUIET >= 2, 1) AND SZ > 0 AND IS_EXPL_SUB AND M_DARK > -0.02;

{==================== 四、主图买入信号 & 四象判定 (与主图一致) ====================}
LLV60 := LLV(L, 60);
HH20  := HHV(H, 20);
HH10C := HHV(C, 10);
HHV3  := HHV(H, 3);
LLV3  := LLV(L, 3);

BODY_SHRINK := ABS(C - O) / (H - L + EPS) < 0.40;
LOW_VOL     := V_PRED < V_MA5 * SHR_R OR V_PRED < V_MA_SAFE * 0.80;

RAW_EXHAUSTION := ((HHV3 - LLV3) / (LLV3 + EPS) * 100 < 3.8 OR BODY_SHRINK)
                AND LOW_VOL AND C >= MA25 * MA25_TOL AND C <= MA25 * 1.025
                AND (C - LLV60) / (LLV60 + EPS) * 100 < 45
                AND (SLP_25 > -0.0001 OR C > MA25);

EXH_CNT5       := COUNT(RAW_EXHAUSTION, 5);
TRY_EXHAUSTION := RAW_EXHAUSTION AND EXH_CNT5 = 1;

DONCHIAN_HIGH := REF(HHV(H, N_TRK), 1);
BREAK_POWER   := (C - DONCHIAN_HIGH) / (ATR_VAL + EPS) > 0.35;
RAW_BUY       := CROSS(C, DONCHIAN_HIGH) AND BREAK_POWER AND STK_OK;

KC_UP    := MA25 + 1.45 * ATR_VAL;
BB_UP    := MA25 + BB_F * STD(C, M_PER);
BB_BREAK := CROSS(C, BB_UP) AND V_PRED > REF(VOL, 1) * 0.7;
H_P1     := IF(N_BAR > 1, REF(H, 1), H);
GAP_SZ   := (L - H_P1) / (H_P1 + EPS) * 100;
MAIN_GAP := GAP_SZ > 0.5 AND C > O AND V_PRED > V_MA5;

BUY_T     := RAW_BUY OR CROSS(C, KC_UP) OR BB_BREAK OR MAIN_GAP;
OPEN_SAFE := NOT(IS_LIVE_ANY) OR FROMOPEN > OPEN_BLOCK;

CORE_COND := BUY_T AND C > MA25 * CORE_TOL AND SLF_TR_UP;
FLT_PASS  := SCALPEL OR MAIN_GAP OR SZ_STR OR (C > HH20 * 0.97 AND VOL_RATIO > 1.0) OR (M_DARK > 0.08) OR M_FLY;

FINAL_BUY := CORE_COND AND FLT_PASS AND MULTI_RESO_OK AND OPEN_SAFE AND WARM_OK;
SIG_STR   := FINAL_BUY AND (TREND_ST OR SZ_VST OR BREAK_POWER);
SIG_NOR   := FINAL_BUY AND NOT(SIG_STR);

TRY_BUY  := TRY_EXHAUSTION AND NOT(FINAL_BUY) AND MULTI_RESO_OK AND OPEN_SAFE AND WARM_OK;
BUY_SIG  := SIG_STR OR SIG_NOR OR TRY_BUY;

{ 棘轮止损 }
HAS_BUY     := SUM(BUY_SIG, 0) > 0;
B_CNT       := IF(HAS_BUY, BARSLAST(BUY_SIG), 9999);
HOLD_LEN    := IF(HAS_BUY, B_CNT + 1, 1);

BUY_REF_C := REF(C, HOLD_LEN - 1);
HH_C_HOLD := HHV(C, HOLD_LEN);
MAX_GAIN  := IF(HAS_BUY, (HH_C_HOLD - BUY_REF_C) / (BUY_REF_C + EPS) * 100, 0);

DYN_X := IF(MAX_GAIN > STRONG_GAIN_PCT OR SZ_SC > SZ_SC_HI, TIGHT_STOP_MULT,
        IF(MAX_GAIN > MID_GAIN_PCT OR SZ_SC > SZ_SC_MID, MID_STOP_MULT,
        IF(TREND_ST AND ADX_STR, ATR_XS, ATR_X)));
TIGHT_LOCK := DYN_X <= TIGHT_STOP_MULT;

RAW_TRAIL    := HH_C_HOLD - DYN_X * ATR_VAL;
STOP_RATCHET := HHV(RAW_TRAIL, HOLD_LEN);

CRASH_EX   := C < STOP_RATCHET OR CROSS(MA25, C) OR C < MA25 * IF(ADX_STR, 0.925, 0.945) OR (B_CNT >= HOLD_MAX);
LAST_CRASH := IF(SUM(CRASH_EX, 0) > 0, BARSLAST(CRASH_EX), 9999);
HOLD_ST    := HAS_BUY AND LAST_CRASH > B_CNT AND B_CNT < HOLD_MAX;

{ 四象 }
IS_SUMMER := HOLD_ST AND (MAX_GAIN > STRONG_GAIN_PCT OR (C >= HH10C AND SZ_SC < SZ_SC_MID) OR TIGHT_LOCK);
IS_SPRING := HOLD_ST AND NOT(IS_SUMMER);
IS_WINTER := NOT(HOLD_ST) AND (TRY_BUY OR RAW_EXHAUSTION) AND WARM_OK;
IS_AUTUMN := NOT(HOLD_ST) AND NOT(IS_WINTER);

BUY_ANY       := SIG_STR OR SIG_NOR;
HOLD_CROSS_DN := HOLD_ST AND M_CROSS_DN;


{==================== 五、副图成交额动量引擎 (SZ_VAL 体系, 仅用于画线/柱高) ====================}
AMT_MA      := IF(N_BAR >= VOL_DAYS, MA(AMT_PRED_RAW, VOL_DAYS), SUM(AMT_PRED_RAW, 0) / N_BAR);
AMT_MA_PRV  := IF(N_BAR > 1, REF(AMT_MA, 1), AMT_MA);
AMT_MA_SAFE := MAX(AMT_MA, MAX(AMT_MA_PRV, AMT_PRED_RAW * 0.12));

PCT_STD     := IF(N_BAR >= 20, STD(PCT_CHG, 20),
                SQRT(MAX(SUM(PCT_CHG * PCT_CHG, 0) / N_BAR - POW(SUM(PCT_CHG, 0) / N_BAR, 2), 0)));
PCT_STD_PRV := IF(N_BAR > 1, REF(PCT_STD, 1), 0);
MAG_BASE    := MAX(MAG_FLOOR, PCT_STD_PRV);

EXPL_TH_BASE := 1.85 - W_ER * 0.60;
EXPL_RATIO   := MIN(AMT_PRED_RAW / (AMT_MA_SAFE + EPS), EX_RATIO_CAP);
EXPL_GAP     := EXPL_RATIO - EXPL_TH_BASE;

SAFE_EXP_IN := MAX(-50, MIN(50, -SMOOTH_K * EXPL_GAP));
EXPL_MULT   := EXPL_BASE_MUL + EXPL_RNG_MUL / (1 + EXP(SAFE_EXP_IN));

DARK_FLOW_RAW := IF(C >= PREV_C, RL_DIR * (0.65 + 0.35 * EXPL_RATIO), RL_DIR * (0.65 + 0.35 * EXPL_RATIO) * SZ_DOWN_COEF);
DARK_FLOW     := EMA(DARK_FLOW_RAW, 3);

VOL_DRIVE_CHK := MIN(1.5, MAX(0.5, AMT_PRED_RAW / (AMT_MA + EPS)));
CZB_DIR := IF(DARK_FLOW > DARK_TH, 1, IF(DARK_FLOW < -DARK_TH, -1, IF(C >= PREV_C, 0.5, -0.5)));

CZB := CZB_DIR * (WEIGHT * EXPL_RATIO * EXPL_MULT * VOL_DRIVE_CHK)
        * MAX(MAG_LO, MIN(MAG_HI, ABS(PCT_CHG) / (MAG_BASE + EPS)));

CZB_SM5  := EMA(CZB, 5);
SZ_OSC_F := EMA(CZB_SM5, 5) - EMA(CZB_SM5, 13);
SZ_OSC_S := EMA(CZB_SM5, 10) - EMA(CZB_SM5, 26);
SZ_VAL   := EMA(W_ER * SZ_OSC_F + (1 - W_ER) * SZ_OSC_S, 5) * SZ_SCALE;

SZ_PRV  := IF(N_BAR > 1, REF(SZ_VAL, 1), SZ_VAL);
D1_RAW  := (SZ_VAL - SZ_PRV) * TIME_DAMP;
D1_PRV  := IF(N_BAR > 1, REF(D1_RAW, 1), D1_RAW);
D2_ACC  := D1_RAW - D1_PRV;

PRV_D2_ACC  := IF(N_BAR > 2, REF(D2_ACC, 1), 0);
ACC_BOOST   := MAX(0, PRV_D2_ACC);
EXPL_TH     := EXPL_TH_BASE - MIN(0.25, ACC_BOOST * 0.5);

LEAD_RAW    := (D1_RAW + D2_ACC * LEAD_D2W) * (LEAD_GAIN * LEAD_DAMP);
LEAD_SMOOTH := EMA(LEAD_RAW, LEAD_SM_LEN);
LEAD_FACTOR := LEAD_RAW * LEAD_ALPHA + LEAD_SMOOTH * (1 - LEAD_ALPHA);
SZ_MA_VAL   := EMA(SZ_VAL + LEAD_FACTOR * LEAD_MIX, SZ_MA_LEN);

SQZ_STD   := IF(N_BAR >= SQZ_N, STD(SZ_VAL, SQZ_N),
                SQRT(MAX(SUM(SZ_VAL * SZ_VAL, 0) / N_BAR - POW(SUM(SZ_VAL, 0) / N_BAR, 2), 0)));
SQZ_MID   := IF(N_BAR >= SQZ_N, MA(SZ_VAL, SQZ_N), SUM(SZ_VAL, 0) / N_BAR);
SQZ_WIDTH := SQZ_STD * 2 / (ABS(SQZ_MID) + 0.01);
SQZ_MIN   := IF(N_BAR >= SQZ_LB, LLV(SQZ_WIDTH, SQZ_LB), LLV(SQZ_WIDTH, 0));

IS_SQUEEZE := N_BAR >= SQZ_WARM AND (SQZ_WIDTH < SQZ_MIN * SQZ_MULT OR SQZ_WIDTH < SQZ_ABS);
SQZ_FIRE   := IS_SQUEEZE AND D1_RAW > 0;

UP_ARR := IF(SZ_VAL > 0, SZ_VAL, 0);
DN_ARR := IF(SZ_VAL < 0, ABS(SZ_VAL), 0);
SUM_UP := IF(N_BAR >= ENERGY_PERIOD, SUM(UP_ARR, ENERGY_PERIOD), SUM(UP_ARR, 0));
SUM_DN := IF(N_BAR >= ENERGY_PERIOD, SUM(DN_ARR, ENERGY_PERIOD), SUM(DN_ARR, 0));
ENERGY_RATIO := IF(SUM_DN = 0, 15, MIN(MAX(SUM_UP / (SUM_DN + EPS), 0.1), 15));

TMC_BASE_F := EMA(SZ_VAL, 7);
TMC_BASE_S := EMA(SZ_VAL, 25);
TMC_BASE   := W_ER * TMC_BASE_F + (1 - W_ER) * TMC_BASE_S;
TMC_PRV    := IF(N_BAR > 1, REF(TMC_BASE, 1), TMC_BASE);
TMC_D1     := (TMC_BASE - TMC_PRV) * TIME_DAMP;
TMC_D1P    := IF(N_BAR > 1, REF(TMC_D1, 1), TMC_D1);
TMC_D2     := TMC_D1 - TMC_D1P;
TMC_LEAD_RAW := (TMC_D1 + TMC_D2 * TMC_D2W) * (TMC_GAIN * TMC_DAMP);
TMC_LEAD_FAC := TMC_LEAD_RAW * TMC_ALPHA + EMA(TMC_LEAD_RAW, TMC_LD_SM_LEN) * (1 - TMC_ALPHA);
TMC_OBSERVE  := EMA(TMC_BASE + TMC_LEAD_FAC, TMC_OUT_LEN);

SZ_CROSS_DN := CROSS(TMC_OBSERVE, SZ_MA_VAL);

{ 副图自己的起爆提示 (仅用于显示"距起爆点", 不参与买入/持仓判定) }
IS_EXPL     := AMT_PRED_RAW > AMT_MA_SAFE * EXPL_TH;
CF_TREND_OK := SZ_MA_VAL > TMC_OBSERVE;
Q_ARR       := IF((SZ_VAL > 0 AND NOT(IS_EXPL)) OR IS_SQUEEZE, 1, 0);
QUIET_SCORE := IF(N_BAR >= 3, SUM(Q_ARR, 3), SUM(Q_ARR, 0));
QUIET_PRV   := IF(N_BAR > 1, REF(QUIET_SCORE >= 2, 1), 0);
