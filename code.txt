             {================ 天道围猎·量能 ================}

{━━━━━━━━━━━━━━━━ 一、环境与参数 ━━━━━━━━━━━━━━━━}
IS_INTRA:=PERIOD<3;
IS_INDEX:=CAPITAL=0 OR C=INDEXC;
SKIP_COST:=IS_INDEX OR IS_INTRA;

N_TRK:=IF(IS_INTRA,5,8);
ATR_X:=1.05;
ATR_XS:=1.42;
ATR_PCT_HI:=0.05;
ATR_PCT_LO:=0.03;

V_DYS:=IF(IS_INTRA,5,8);
SHR_R:=0.82;
EX_CAP:=6;
EXPL_TH_MIN_AMT:=0.35;

DARK_TH:=0.045;
ENERGY_PRD:=IF(IS_INTRA,20,30);

M_PER:=IF(IS_INTRA,5,8);
BB_F:=IF(IS_INTRA,1.5,2.0);

HOLD_MAX:=IF(IS_INTRA,48,240);
HOLD_LEN_MIN:=5;
STR_GAIN_PCT:=20;
TGT_STP_MULT:=0.68;
STP_FLOOR_B:=0.75;
STP_FLOOR_C:=0.72;
MA21_STP_FLR:=0.945;
STP_RTCH_BUF:=0.998;
EXIT_COOL_N:=8;

SZ_SC_MID:=0.90;
WEIGHT:=3.45;
SMOOTH_K:=5.5;
LEAD_GAIN:=2.55;
LEAD_RAW_W:=0.55;
SZ_MA_LEN:=IF(IS_INTRA,12,9);
SZ_SC_OFF:=1.9;
SZ_SC_SCL:=0.37037;
PEAK_TH_BS:=0.30;
VALLEY_TH_BS:=-0.30;

TMC_GAIN:=2.35;
TMC_ALPHA:=0.60;
TMC_LD_SM:=IF(IS_INTRA,5,3);
TMC_OUT_BS:=10.5;
SQZ_FLR_MLT:=1.05;

FLT_MIN_SCR:=4;
CHIP_SCR_TH:=11.0;
VWAP_UP_LIM:=1.045;
STP_PRV_MLT:=5;
PEAK_COOL:=IF(IS_INTRA,5,8);

EPS:=0.0001;
ER_PERIOD:=8;

{━━━━━━━━━━━━━━━━ 二、基础数据（合并重复引用）━━━━━━━━━━━━━━━━}
PREV_C:=REF(C,1);
PREV_H:=REF(H,1);
PREV_L:=REF(L,1);
C_E:=C+EPS;
RNG_HL:=H-L;
RNG_HL_E:=RNG_HL+EPS;
ABS_CHG:=ABS(C-PREV_C);
{TR_BASE 三取最大，合并为一次计算}
TR_BASE:=MAX(MAX(RNG_HL,ABS(H-PREV_C)),ABS(L-PREV_C));
BARS_CNT:=BARSCOUNT(C);
BARS_WARM:=BARS_CNT>ENERGY_PRD;

{━━━━━━━━━━━━━━━━ 三、KAUFMAN ER ━━━━━━━━━━━━━━━━}
GAP_COND:=(L>PREV_H) OR (H<PREV_L);
C90_HIGH:=IF(SKIP_COST,H,COST(95));
C90_LOW:=IF(SKIP_COST,L,COST(5));
C90_MID:=(C90_HIGH+C90_LOW)*0.5;
C90_SPREAD:=C90_HIGH-C90_LOW;
{HAS_COST：将除法判断折叠，减少分支}
HAS_COST:=IF(SKIP_COST,0,C90_HIGH>EPS AND C90_MID>EPS AND ABS(C90_SPREAD)/(C90_MID+EPS)>0.0005);

SUM_ABS_CHG8:=SUM(ABS_CHG,ER_PERIOD);
W_ER_RAW:=ABS(C-REF(C,ER_PERIOD))/(MAX(SUM_ABS_CHG8,C*0.001)+EPS);
W_ER:=MIN(MAX(W_ER_RAW,GAP_COND*0.90),1);
W_ER_INV:=1-W_ER;
TMC_ER_LEN:=MAX(4,ROUND(8-4*W_ER));

{━━━━━━━━━━━━━━━━ 四、筹码与 VWAP（减少冗余乘除）━━━━━━━━━━━━━━━━}
SCR90:=MIN(MAX(IF(HAS_COST,C90_SPREAD/(C90_MID+EPS)*100,0),0),100);
WIN_RATIO:=IF(SKIP_COST,50,MIN(MAX(WINNER(C)*100,0),100));
CHIP_LOCK:=IF(SKIP_COST,0,HAS_COST AND SCR90<CHIP_SCR_TH AND WIN_RATIO>55);
CHIP_EFF:=IF(HAS_COST,MIN(MAX(0.6+0.4*(1-SCR90*0.008),0.6),1.0),0.8);

TOT_AMT:=SUM(AMOUNT,8);
TOT_VOL_RAW:=SUM(VOL,8);
{合并 TOT_VOL_SAFE，避免 MAX+EPS 分拆}
TOT_VOL_SAFE:=(MAX(TOT_VOL_RAW,C)+EPS)*100;
VWAP_RAW:=TOT_AMT/TOT_VOL_SAFE;
{C_E 已定义，直接复用}
VWAP_VALID:=TOT_VOL_RAW>0 AND VWAP_RAW>C*0.95 AND VWAP_RAW<C*VWAP_UP_LIM;
VWAP_SUPP_VAL:=IF(VWAP_VALID,VWAP_RAW,C);
VWAP_SUPP:=VWAP_VALID AND C>=VWAP_SUPP_VAL*0.98 AND C<=VWAP_SUPP_VAL*1.02;

{━━━━━━━━━━━━━━━━ 五、自适应 ATR 与趋势生命线（减少 MA 层）━━━━━━━━━━━━━━━━}
ATR_FAST:=MA(TR_BASE,5);
TR_MA21:=MA(TR_BASE,10);
ATR_SLOW:=MA(TR_BASE,18);
{W_ER_INV 已缓存，直接用}
ATR_VAL:=W_ER*ATR_FAST+W_ER_INV*(0.6*TR_MA21+0.4*ATR_SLOW);
ATR_VAL_E:=ATR_VAL+EPS;
ATR_PCT:=ATR_VAL/C_E;
ATR_ADAPT:=0.90+(ATR_PCT>ATR_PCT_LO)*0.10+(ATR_PCT>ATR_PCT_HI)*0.15;

MA21:=EMA(C,M_PER);
MA21_R1:=REF(MA21,1);
SYS_MA55:=EMA(C,IF(IS_INTRA,13,34));
MA55_R1:=REF(SYS_MA55,1);
MA55_R2:=REF(MA55_R1,1);

{MA21 倍数：集中一处计算，后续直接引用}
MA21_985:=MA21*0.985;
MA21_992:=MA21*0.992;
MA21_945:=MA21*MA21_STP_FLR;
MA21_018:=MA21*1.018;
MA21_E:=MA21+EPS;
SLP_21:=(MA21-MA21_R1)/MAX(M_PER,1)/MA21_E;
TREND_ST:=SLP_21>0.00030;
STK_OK:=C>MA21_985;
SLF_TR_UP:=MA55_R1>=MA55_R2 OR C>SYS_MA55;
STD_C_MPER:=STD(C,M_PER);
BB_UPPER:=MA21+BB_F*STD_C_MPER;

{━━━━━━━━━━━━━━━━ 六、DMI 与 ADX（缓存 TR21_I100）━━━━━━━━━━━━━━━━}
HD:=H-PREV_C;
LD:=PREV_L-L;
DMP_RAW:=(HD>LD AND HD>0)*HD;
DMM_RAW:=(LD>HD AND LD>0)*LD;
TR21_I100:=100/(TR_MA21+EPS);
PDI_S_ADX:=MA(DMP_RAW,10)*TR21_I100;
MDI_S_ADX:=MA(DMM_RAW,10)*TR21_I100;
DI_SUM_E:=MDI_S_ADX+PDI_S_ADX+EPS;
DI_DIFF:=ABS(MDI_S_ADX-PDI_S_ADX);
ADX_VAL:=EMA(DI_DIFF/DI_SUM_E*100,10);
ADX_STR_S:=ADX_VAL>25;

{━━━━━━━━━━━━━━━━ 七、跨周期趋势（日线引用减少重复）━━━━━━━━━━━━━━━━}
EMA21_CRS:=IF(IS_INTRA,EMA(C,10),MA21);
EMA55_CRS:=IF(IS_INTRA,EMA(C,34),SYS_MA55);
TD_INTRA_BS:=IF(MA21>SYS_MA55,35,0)
             +IF(MA21>MA21_R1,25,10*(MA21=MA21_R1))
             +IF(SYS_MA55>MA55_R1,20,0)
             +IF(C>MA21,20,0);

D_C:=REF(C,1)#DAY;
D_M21:=REF(EMA21_CRS,1)#DAY;
D_M55:=REF(EMA55_CRS,1)#DAY;
D_M21_PR:=REF(EMA21_CRS,2)#DAY;
D_M55_PR:=REF(EMA55_CRS,2)#DAY;
TD_DAILY_BS:=IF(D_M21>D_M55,35,0)
             +IF(D_M21>D_M21_PR,25,10*(D_M21>=D_M21_PR*0.9999))
             +IF(D_M55>D_M55_PR,20,0)
             +IF(D_C>D_M21,20,0);
TREND_DAY:=IF(IS_INTRA,TD_INTRA_BS,TD_DAILY_BS);
MULTI_RESO:=TREND_DAY>=55 AND SLF_TR_UP AND SLP_21>-0.001;
TD_STR:=TREND_DAY>=80 AND ADX_STR_S AND TREND_ST AND C>SYS_MA55;

{━━━━━━━━━━━━━━━━ 八、动量量能预测（合并冗余 EMA 调用）━━━━━━━━━━━━━━━━}
SAFE_WIN21:=MIN(V_DYS,MAX(BARS_CNT,3));
CUR_MIN:=MAX(FROMOPEN,1);
IS_LB_IN:=ISLASTBAR AND IS_INTRA;
BAR_MIN_RAW:=MOD(CUR_MIN,10);
BAR_MIN:=IF(BAR_MIN_RAW=0,10,BAR_MIN_RAW);
SCALE_SAFE:=MIN(240/MAX(BAR_MIN,5),8);
LB_IN_COND:=IS_LB_IN AND BAR_MIN>=5 AND CUR_MIN<240;

AMT_PRD_RW:=IF(LB_IN_COND,AMOUNT*SCALE_SAFE,AMOUNT);
AMT_PRED_RW:=IF(IS_INTRA,EMA(AMT_PRD_RW,5),AMT_PRD_RW);
AMT_MA21:=MA(AMT_PRD_RW,SAFE_WIN21);
AMT_MA21_SF:=MAX(AMT_MA21,MAX(REF(AMT_MA21,1),AMT_PRED_RW*EXPL_TH_MIN_AMT))+EPS;

PCT_CHG:=(C-PREV_C)/(PREV_C+EPS)*100;
ABS_PCT_CHG:=ABS(PCT_CHG);
STD_PCT_W21:=STD(PCT_CHG,SAFE_WIN21);
MAG_BASE:=MAX(0.35,REF(STD_PCT_W21,1));
EXPL_TH:=1.80-W_ER*0.55;
EXPL_DEN:=EXPL_TH-0.8+EPS;

VOL_PRD_RW:=IF(LB_IN_COND,VOL*SCALE_SAFE,VOL);
VOL_PRED:=IF(IS_INTRA,EMA(VOL_PRD_RW,5),VOL_PRD_RW);
V_MA:=MA(VOL,V_DYS);
V_MA3:=MA(VOL,3);
V_MA5:=MA(VOL,5);
V_MA8:=MA(VOL,8);
{V_MA_SGE / V_MA_PRE 共用 MAX 基础，合并同类分母}
V_MA_SGE:=MAX(V_MA,VOL*0.15)+EPS;
V_MA_PRE:=MAX(V_MA,VOL_PRED*0.15)+EPS;
V_SHRK:=VOL<V_MA_SGE*SHR_R;
EXPL_RATIO:=MIN(AMT_PRED_RW/AMT_MA21_SF,EX_CAP);
EXPL_GAP:=EXPL_RATIO-EXPL_TH;
EXPL_GT1:=EXPL_RATIO>1.0;
IS_EXPL_SUB:=EXPL_RATIO>EXPL_TH;
{EXP 函数只算一次}
EXPL_MULT:=1.28+1.22/(1+EXP(-SMOOTH_K*EXPL_GAP));
MAG_BASE_E:=MAG_BASE+EPS;
MAG_RAT_N:=ABS_PCT_CHG/MAG_BASE_E;
MAG_W_TRD:=MIN(W_ER*1.8,1.0);
VOL_RATIO:=MIN(VOL_PRED/V_MA_PRE,EX_CAP);

{VOL_LEAD：去掉双重 EMA，只保留一层}
VOL_LEAD_RAW:=EMA(VOL_RATIO/(MAG_RAT_N+1.0),5)*0.35;
VOL_LEAD_N:=MIN(MAX(VOL_LEAD_RAW,0.2),1.8);
MAG_CLAMP:=MAX(0.5,MIN(2.0,MAG_W_TRD*MAX(0.5,MIN(2.0,MAG_RAT_N))+(1-MAG_W_TRD)*VOL_LEAD_N));

{━━━━━━━━━━━━━━━━ 九、暗流引擎（提取公共子式）━━━━━━━━━━━━━━━━}
IS_LIMIT_UP:=H=L AND C>PREV_C;
IS_LIMIT_DN:=H=L AND C<PREV_C;
INTRA_DIRB:=IF(RNG_HL>EPS,((C-L)-(H-C))/RNG_HL,0);
INTRA_DIRR:=IF(IS_LIMIT_UP,1.0,IF(IS_LIMIT_DN,-1.0,INTRA_DIRB));

{DARK_FLW_R 内联 EXPL_RATIO，避免重复乘法}
DARK_FLW_R:=INTRA_DIRR*0.50*(0.60+0.40*EXPL_RATIO)*CHIP_EFF;
DARK_FLW_R1:=REF(DARK_FLW_R,1);
DARK_FLOW:=EMA(DARK_FLW_R,IF(IS_INTRA,6,4));
DARK_ACCEL:=EMA((DARK_FLW_R-DARK_FLW_R1)*1.20,IF(IS_INTRA,8,3));

DF_REF1:=REF(DARK_FLOW,1);
DF_OK_SOFT:=DARK_FLOW>-0.02;
{CZB_DIR 三分支合并，减少 IF 嵌套层数}
CZB_DIR:=(DARK_FLOW>DARK_TH)*1
         +(DARK_FLOW<-DARK_TH)*(-1)
         +(DARK_FLOW>=-DARK_TH AND DARK_FLOW<=DARK_TH)*(IF(C>=PREV_C,0.5,-0.5));
CZB_CORE:=WEIGHT*EXPL_RATIO*EXPL_MULT;
CZB:=CZB_DIR*CZB_CORE*MAG_CLAMP;

CZB_SM5:=EMA(CZB,IF(IS_INTRA,8,5));
{四条 EMA 从 CZB_SM5 展开，源头只算一次}
CZB_EMA5:=EMA(CZB_SM5,5);
CZB_EMA8:=EMA(CZB_SM5,8);
CZB_EMA10:=EMA(CZB_SM5,10);
CZB_EMA20:=EMA(CZB_SM5,20);
SZ_OSC_F:=CZB_EMA5-CZB_EMA10;
SZ_OSC_S:=CZB_EMA8-CZB_EMA20;
SZ_RAW:=EMA(W_ER*SZ_OSC_F+W_ER_INV*SZ_OSC_S,IF(IS_INTRA,5,3))*0.75*CHIP_EFF;
SZ:=MAX(MIN(SZ_RAW+DARK_ACCEL*0.65,15),-15);
SZ_REF1:=REF(SZ,1);
D1_RAW:=SZ-SZ_REF1;
MOM_BOOST:=1.0+(PCT_CHG>0)*PCT_CHG*0.10*W_ER*1.2;

SZ_ADJ_LEN:=MAX(IF(IS_INTRA,8,5),INTPART(SZ_MA_LEN*(1-W_ER)+6*W_ER+0.5));
LEAD_RAW_A:=MIN(MAX((D1_RAW+DARK_ACCEL*0.35)*LEAD_GAIN*MOM_BOOST,-8.5),8.5);
LEAD_SMOOTH:=EMA(LEAD_RAW_A,SZ_ADJ_LEN);
LEAD_FACTOR:=LEAD_RAW_A*LEAD_RAW_W+LEAD_SMOOTH*(1-LEAD_RAW_W);
SZ_MA_VAL:=MAX(MIN(EMA(SZ+LEAD_FACTOR*0.8,SZ_ADJ_LEN),999),-999);
SZ_MA_R1:=REF(SZ_MA_VAL,1);
SZ_MA_R2:=REF(SZ_MA_R1,1);

SZ_P_UP:=SZ_MA_VAL>SZ_MA_R1 AND DARK_FLOW>0.008 AND W_ER>0.32;
SZ_HIGH_PUR:=SZ_P_UP AND SZ_MA_VAL>0.25;

{SQZ：STD 和 LLV 只算一次}
SZ_MA21:=MA(SZ,10);
SQZ_STD:=STD(SZ,8);
SQZ_DENOM:=ABS(SZ_MA21)+0.01;
SQZ_WIDTH:=SQZ_STD*2/SQZ_DENOM;
SQZ_MIN:=LLV(SQZ_WIDTH,20)*SQZ_FLR_MLT;
IS_SQUEEZE:=SQZ_WIDTH<SQZ_MIN;

{DIR_VOL：复用 V_MA_SGE，减少一次除法}
DIR_VOL_RW:=IF(C>PREV_C,VOL,IF(C<PREV_C,-VOL,VOL*INTRA_DIRR));
DIR_VOL_AC:=EMA(DIR_VOL_RW/V_MA_SGE,8);
DIR_VOL_NM:=MIN(MAX(EMA(DIR_VOL_AC,5),-2.5),2.5);
DIR_VOL_ABS:=ABS(DIR_VOL_RW);
VOL_SUM10:=SUM(VOL,10);
VOL_SUM_EP:=VOL_SUM10+EPS;
VOL_ACCUM:=SUM(DIR_VOL_RW,10);
VOL_ACC_POS:=SUM(IF(DIR_VOL_RW>0,DIR_VOL_RW,0),10);
VOL_ACC_MA:=MA(DIR_VOL_ABS,10)+EPS;
VOL_ACC_EP:=VOL_ACC_MA*10;
ENERGY_CHG:=EMA(MIN(MAX(VOL_ACCUM/VOL_ACC_EP,-1),1),5);
DIR_CONSIST:=VOL_ACC_POS/VOL_SUM_EP>0.55;
ENERGY_RDY:=BARS_WARM AND ENERGY_CHG>0.20 AND EXPL_GT1 AND DIR_CONSIST;

{━━━━━━━━━━━━━━━━ 十、评分与 TMC（权重系数预乘）━━━━━━━━━━━━━━━━}
W_S1:=0.20+W_ER*0.20;
W_S2:=0.35-W_ER*0.15;
W_S3:=0.30-W_ER*0.10;
W_S4:=0.15+W_ER*0.15;
DF_NORM:=MIN(MAX((DARK_FLOW+0.05)/0.10,0),1.0);

{ENERGY_SCR：将四项分量先算好再加权，结构更清晰，编译器更易优化}
ESC_P1:=MIN(MAX((EXPL_RATIO-1.0)/EXPL_DEN,0),1.0)*W_S1;
ESC_P2:=MIN(MAX(ENERGY_CHG,0),1.0)*IF(DIR_CONSIST,1.0,0.6)*W_S2;
ESC_P3:=MIN(MAX(MIN(VOL_RATIO,3.0)/3.0,0),1.0)*IF(V_SHRK,0.8,1.0)*W_S3;
ESC_P4:=DF_NORM*CHIP_EFF*W_S4;
ENERGY_SCR:=MIN(ROUND((ESC_P1+ESC_P2+ESC_P3+ESC_P4)*100),100);
ENG_VAL_OK:=ENERGY_SCR>=66;
ENG_EXPLODE:=ENERGY_SCR>=82;
HI_RES_ENG:=ENERGY_SCR>=80;

TMC_OUT_LN:=MAX(IF(IS_INTRA,8,5),INTPART(TMC_OUT_BS*(1-W_ER*0.50)+0.5));
TMC_BASE_F:=EMA(SZ,TMC_ER_LEN);
TMC_BASE_S:=EMA(SZ,IF(IS_INTRA,13,20));
TMC_BASE:=W_ER*TMC_BASE_F+W_ER_INV*TMC_BASE_S;
TMC_BS_LED:=TMC_BASE+DIR_VOL_NM*0.18*(1-W_ER);
TMC_BS_R1:=REF(TMC_BS_LED,1);
TMC_D1:=TMC_BS_LED-TMC_BS_R1;
TMC_D1_R1:=REF(TMC_D1,1);
TMC_LD_RAW:=MIN(MAX((TMC_D1+(TMC_D1-TMC_D1_R1)*0.55)*TMC_GAIN,-5.8),5.8);
TMC_LD_EMA3:=EMA(TMC_LD_RAW,TMC_LD_SM);
{TMC_1MA 提前缓存，避免反复 1-TMC_ALPHA}
TMC_1MA:=1-TMC_ALPHA;
TMC_LD_FAC:=TMC_LD_RAW*(TMC_ALPHA+W_ER*TMC_1MA)+TMC_LD_EMA3*(W_ER_INV*TMC_1MA);
TMC_OBSERVE:=MAX(MIN(EMA(TMC_BS_LED+TMC_LD_FAC,TMC_OUT_LN),999),-999);
TMC_OBS_R1:=REF(TMC_OBSERVE,1);
TMC_SLP_UP:=TMC_OBSERVE>TMC_OBS_R1;
SZ_POS:=SZ>0;

QUIET_COND:=(SZ_POS AND NOT(IS_EXPL_SUB)) OR IS_SQUEEZE;
QUIET_SCR:=COUNT(QUIET_COND,5);
CF_FLY_SIG:=SZ_MA_VAL>TMC_OBSERVE AND QUIET_SCR>=2 AND SZ_POS AND IS_EXPL_SUB AND DF_OK_SOFT AND TMC_SLP_UP AND SZ_HIGH_PUR;

DF_CROSS_DN:=CROSS(TMC_OBSERVE,SZ_MA_VAL) AND HHV(ADX_VAL,10)-ADX_VAL>3;
SZ_SC:=MIN(MAX((SZ+SZ_SC_OFF)*SZ_SC_SCL,0),2.0);
SZ_ACC_S:=D1_RAW>0 AND SZ>-0.85;
SZ_VST:=SZ_SC>1.15;
SZ_STR:=SZ_SC>0.72;
SCALPEL:=V_SHRK AND SZ>-1.0 AND SZ_ACC_S;

{SZ_MA_STD60：STD 只算一次，复用}
SZ_MA_STD60:=MAX(STD(SZ_MA_VAL,20)*0.8,0.15);
PEAK_TH:=MAX(PEAK_TH_BS,SZ_MA_STD60);
VALLEY_TH:=MIN(VALLEY_TH_BS,-SZ_MA_STD60);

{━━━━━━━━━━━━━━━━ 公共计算 ━━━━━━━━━━━━━━━━}
HH_PRD:=HHV(H,15);
LL_PRD:=LLV(L,15);
RNG_PRD:=HH_PRD-LL_PRD+EPS;
PRC_POS:=(C-LL_PRD)/RNG_PRD;
BIAS21:=(C-MA21)/MA21_E*100;

{顶底反转：修正冷却距离判定（采用上一次到当前的真实间隔）}
SZ_PK_BS:=SZ_MA_R1>SZ_MA_R2 AND SZ_MA_R1>=SZ_MA_VAL AND SZ_MA_R1>PEAK_TH;
DPK_PREV:=REF(BARSLAST(SZ_PK_BS),1)+1;
SZ_PEAK:=SZ_PK_BS AND (PRC_POS>=0.65 OR BIAS21>3.5 OR DARK_FLOW<DF_REF1) AND (DPK_PREV>PEAK_COOL OR BARS_CNT<12);

SZ_VL_BS:=SZ_MA_R1<SZ_MA_R2 AND SZ_MA_R1<=SZ_MA_VAL AND SZ_MA_R1<VALLEY_TH;
DVL_PREV:=REF(BARSLAST(SZ_VL_BS),1)+1;
SZ_VALLEY:=SZ_VL_BS AND (PRC_POS<=0.55 OR BIAS21<2.0) AND (DVL_PREV>PEAK_COOL OR BARS_CNT<12);

{━━━━━━━━━━━━━━━━ 十一、VCP ━━━━━━━━━━━━━━━━}
VCP_HH20:=HHV(H,10);
VCP_LL20:=LLV(L,10);
VCP_LL20_E:=VCP_LL20+EPS;
AMP_T1:=(VCP_HH20-VCP_LL20)/VCP_LL20_E*100;

TS_HHV10:=HHV(H,8);
TS_LLV10:=LLV(L,8);
TS_LLV10_E:=TS_LLV10+EPS;
AMP_T2:=(TS_HHV10-TS_LLV10)/TS_LLV10_E*100;
