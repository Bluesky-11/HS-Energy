{==================================================================
            天道围猎·资金气数动量图 (道友修复优化版)
==================================================================}

{==================== 一、自适应预热与无量纲化隔离 ====================}
VOL_DAYS      := 10;
WEIGHT        := 3.55;
SMOOTH_K      := 6.8;

{ --- 主动量超前参数 (SZ_MA_LINE) --- }
LEAD_GAIN     := 2.95;
LEAD_ALPHA    := 0.85;
LEAD_SM_LEN   := 1;
SZ_MA_LEN     := 2;

{ --- TMC 物理超前参数 (TMC_OBS_LINE) --- }
TMC_GAIN      := 2.35;
TMC_ALPHA     := 0.82;
TMC_LD_SM_LEN := 2;
TMC_OUT_LEN   := 2;

ENERGY_PERIOD := 66;
EX_RATIO_CAP  := 8;
EPS           := 0.0001;
MA20_TOL      := 0.985;
CORE_TOL      := 0.982;
EXIT_TOL      := 0.945;
SEASON_BAR_W  := 0.35;
DARK_TH       := 0.05;

V_DYS_M       := 15;
TRD_MIN       := 240;
PRED_BAR_LEN  := 1;        { 1分钟图=1, 5分钟图=5, 日线=240 }
WARM_N        := 10;       { 买点预热根数 }
SQZ_WARM      := 20;       { 挤压判定所需最小根数 }

N_BAR   := BARSCOUNT(C);   { 当前是第几根 }
WARM_OK := N_BAR > WARM_N;

{ 外推系数 }
CUR_MIN := MAX(FROMOPEN, 1);
MM_BAR  := MOD(CUR_MIN, PRED_BAR_LEN);
BAR_MIN := IF(MM_BAR = 0, PRED_BAR_LEN, MM_BAR);
IS_PRED := ISLASTBAR AND CUR_MIN < TRD_MIN AND PERIOD < 5;
PRED_F  := IF(IS_PRED, PRED_BAR_LEN / MAX(BAR_MIN, PRED_BAR_LEN / 8), 1);

AMT_PRED_RAW := AMOUNT * PRED_F;
V_PRED       := VOL * PRED_F;

{ 前收与首根处理 (优化: 使用 VALUEWHEN 代替旧式累加) }
PC1     := IF(N_BAR > 1, REF(C, 1), O);
PREV_C  := MAX(PC1, 0.01);
PCT_CHG := (C - PREV_C) / PREV_C * 100;
FIRST_C := VALUEWHEN(N_BAR = 1, C);

{ 均额: 不足 VOL_DAYS 根时用当前累计均值 }
AMT_MA      := IF(N_BAR >= VOL_DAYS, MA(AMT_PRED_RAW, VOL_DAYS), SUM(AMT_PRED_RAW, N_BAR) / N_BAR);
AMT_MA_PRV  := IF(N_BAR > 1, REF(AMT_MA, 1), AMT_MA);
AMT_MA_SAFE := MAX(AMT_MA, MAX(AMT_MA_PRV, AMT_PRED_RAW * 0.12));

{ 波动基准 (修正: 替换非法 SUM 周期为 N_BAR) }
PCT_STD     := IF(N_BAR >= 20, STD(PCT_CHG, 20),
                SQRT(MAX(SUM(PCT_CHG * PCT_CHG, N_BAR) / N_BAR - POW(SUM(PCT_CHG, N_BAR) / N_BAR, 2), 0)));
PCT_STD_PRV := IF(N_BAR > 1, REF(PCT_STD, 1), 0);
MAG_BASE    := MAX(0.25, PCT_STD_PRV);

{ 核心 ER }
NET_CHG  := IF(N_BAR > 10, ABS(C - REF(C, 10)), ABS(C - FIRST_C));
PATH_SUM := IF(N_BAR >= 10, SUM(ABS(C - PC1), 10), SUM(ABS(C - PC1), N_BAR));
ER_SUB   := IF(PATH_SUM < EPS, 0.5, MIN(NET_CHG / (PATH_SUM + EPS), 1));
W_ER     := EMA(ER_SUB, 5);

EXPL_TH  := 1.85 - W_ER * 0.60;

{==================== 二、暗流控盘 & CZB 动量 ENGINE ====================}
EXPL_RATIO := MIN(AMT_PRED_RAW / (AMT_MA_SAFE + EPS), EX_RATIO_CAP);
EXPL_GAP   := EXPL_RATIO - EXPL_TH;

SAFE_EXP_IN := MAX(-50, MIN(50, -SMOOTH_K * EXPL_GAP));
EXPL_MULT   := 1.28 + 1.22 / (1 + EXP(SAFE_EXP_IN));

K_RANGE   := MAX(H - L, MAX(C * 0.002, ABS(C - PC1)));
INTRA_DIR := ((C - L) - (H - C)) / K_RANGE;
REAL_DIR  := IF(ABS(INTRA_DIR) < 0.01, IF(C > PC1, 0.6, IF(C < PC1, -0.6, 0)), INTRA_DIR);

DARK_FLOW_RAW := REAL_DIR * (0.65 + 0.35 * EXPL_RATIO);
DARK_FLOW     := EMA(DARK_FLOW_RAW, 3);

CZB_DIR := IF(DARK_FLOW > DARK_TH, 1, IF(DARK_FLOW < -DARK_TH, -1, IF(C >= PC1, 0.5, -0.5)));

CZB := CZB_DIR * (WEIGHT * EXPL_RATIO * EXPL_MULT)
       * MAX(0.5, MIN(2.0, ABS(PCT_CHG) / (MAG_BASE + EPS)));

CZB_SM5  := EMA(CZB, 3);
SZ_OSC_F := EMA(CZB_SM5, 5) - EMA(CZB_SM5, 13);
SZ_OSC_S := EMA(CZB_SM5, 10) - EMA(CZB_SM5, 26);
SZ_VAL   := EMA(W_ER * SZ_OSC_F + (1 - W_ER) * SZ_OSC_S, 4) * 0.69;

{==================== 三、导数、挤压与自适应观察线 ====================}
SZ_PRV  := IF(N_BAR > 1, REF(SZ_VAL, 1), SZ_VAL);
D1_RAW  := SZ_VAL - SZ_PRV;
D1_PRV  := IF(N_BAR > 1, REF(D1_RAW, 1), D1_RAW);
D2_ACC  := D1_RAW - D1_PRV;

LEAD_RAW    := (D1_RAW + D2_ACC * 0.618) * LEAD_GAIN;
LEAD_SMOOTH := EMA(LEAD_RAW, LEAD_SM_LEN);
LEAD_FACTOR := LEAD_RAW * LEAD_ALPHA + LEAD_SMOOTH * (1 - LEAD_ALPHA);

SZ_MA_VAL := EMA(SZ_VAL + LEAD_FACTOR * 1.25, SZ_MA_LEN);

{ 挤压指标 (修正: 替换非法 SUM) }
SQZ_STD   := IF(N_BAR >= 20, STD(SZ_VAL, 20),
                 SQRT(MAX(SUM(SZ_VAL * SZ_VAL, N_BAR) / N_BAR - POW(SUM(SZ_VAL, N_BAR) / N_BAR, 2), 0)));
SQZ_MID   := IF(N_BAR >= 20, MA(SZ_VAL, 20), SUM(SZ_VAL, N_BAR) / N_BAR);
SQZ_WIDTH := SQZ_STD * 2 / (ABS(SQZ_MID) + 0.01);
SQZ_MIN   := IF(N_BAR >= 40, LLV(SQZ_WIDTH, 40), LLV(SQZ_WIDTH, N_BAR));

IS_SQUEEZE := N_BAR >= SQZ_WARM AND (SQZ_WIDTH < SQZ_MIN * 1.25 OR SQZ_WIDTH < 0.22);
SQZ_FIRE   := IS_SQUEEZE AND D1_RAW > 0;

{ 多空能量 (修正: 替换非法 SUM) }
UP_ARR := IF(SZ_VAL > 0, SZ_VAL, 0);
DN_ARR := IF(SZ_VAL < 0, ABS(SZ_VAL), 0);
SUM_UP := IF(N_BAR >= ENERGY_PERIOD, SUM(UP_ARR, ENERGY_PERIOD), SUM(UP_ARR, N_BAR));
SUM_DN := IF(N_BAR >= ENERGY_PERIOD, SUM(DN_ARR, ENERGY_PERIOD), SUM(DN_ARR, N_BAR));
ENERGY_RATIO := IF(SUM_DN = 0, 15, MIN(MAX(SUM_UP / (SUM_DN + EPS), 0.1), 15));

{ ==================== TMC 超前计算模块 (道友优化版) ==================== }
TMC_BASE_F := EMA(SZ_VAL, 7);
TMC_BASE_S := EMA(SZ_VAL, 25);
TMC_BASE   := W_ER * TMC_BASE_F + (1 - W_ER) * TMC_BASE_S;

TMC_PRV := IF(N_BAR > 1, REF(TMC_BASE, 1), TMC_BASE);
TMC_D1  := TMC_BASE - TMC_PRV;
TMC_D1P := IF(N_BAR > 1, REF(TMC_D1, 1), TMC_D1);
TMC_D2  := TMC_D1 - TMC_D1P;

{ 优化点：适当调小超前增益，增加内部平滑长度，过滤高频毛刺 }
TMC_LEAD_RAW := (TMC_D1 + TMC_D2 * 0.4) * (TMC_GAIN * 0.85); 
TMC_LEAD_FAC := TMC_LEAD_RAW * 0.75 + EMA(TMC_LEAD_RAW, TMC_LD_SM_LEN + 1) * 0.25;
TMC_OBSERVE  := EMA(TMC_BASE + TMC_LEAD_FAC, TMC_OUT_LEN + 1);

{==================== 四、乘风起爆与主图条件镜像 ====================}
IS_EXPL     := AMT_PRED_RAW > AMT_MA_SAFE * EXPL_TH;
CF_TREND_OK := SZ_MA_VAL > TMC_OBSERVE;

Q_ARR       := IF((SZ_VAL > 0 AND NOT(IS_EXPL)) OR IS_SQUEEZE, 1, 0);
QUIET_SCORE := IF(N_BAR >= 3, SUM(Q_ARR, 3), SUM(Q_ARR, N_BAR));
QUIET_PRV   := IF(N_BAR > 1, REF(QUIET_SCORE >= 2, 1), 0);

CF_FLY_SIGNAL := CF_TREND_OK AND QUIET_PRV AND SZ_VAL > 0 AND IS_EXPL AND DARK_FLOW > -0.02;

HAS_FLY     := SUM(CF_FLY_SIGNAL, N_BAR) > 0;
BARS_TO_FLY := IF(HAS_FLY, BARSLAST(CF_FLY_SIGNAL), 0);

MA20_SUB := EMA(C, 20);
MA20_P5  := IF(N_BAR > 5, REF(MA20_SUB, 5), MA20_SUB);
SLP_20   := (MA20_SUB - MA20_P5) / 5 / (MA20_SUB + EPS);

V_MA      := IF(N_BAR >= V_DYS_M, MA(V_PRED, V_DYS_M), SUM(V_PRED, N_BAR) / N_BAR);
V_MA_SAFE := MAX(V_MA, V_PRED * 0.15);
V_MA5     := IF(N_BAR >= 5, MA(V_PRED, 5), SUM(V_PRED, N_BAR) / N_BAR);
LOW_VOL   := V_PRED < V_MA5 * 0.85 OR V_PRED < V_MA_SAFE * 0.80;

LLV3  := IF(N_BAR >= 3, LLV(L, 3), LLV(L, N_BAR));
HHV3  := IF(N_BAR >= 3, HHV(H, 3), HHV(H, N_BAR));
LLV60 := IF(N_BAR >= 60, LLV(L, 60), LLV(L, N_BAR));

RAW_EXHAUSTION := ((HHV3 - LLV3) / (LLV3 + EPS) * 100 < 3.8 OR ABS(C - O) / (H - L + EPS) < 0.40)
                 AND LOW_VOL
                 AND C >= MA20_SUB * MA20_TOL AND C <= MA20_SUB * 1.025
                 AND (C - LLV60) / (LLV60 + EPS) * 100 < 45
                 AND (SLP_20 > -0.0001 OR C > MA20_SUB);

HHV16       := IF(N_BAR >= 16, HHV(H, 16), HHV(H, N_BAR));
DONCHIAN_H  := IF(N_BAR > 1, REF(HHV16, 1), HHV16);
TR_BASE_SUB := MAX(MAX(H - L, ABS(H - PREV_C)), ABS(L - PREV_C));
TR_AVG      := SUM(TR_BASE_SUB, N_BAR) / N_BAR;
TR_MA12     := IF(N_BAR >= 12, MA(TR_BASE_SUB, 12), TR_AVG);
TR_MA24     := IF(N_BAR >= 24, MA(TR_BASE_SUB, 24), TR_AVG);
TR_MA40     := IF(N_BAR >= 40, MA(TR_BASE_SUB, 40), TR_AVG);
ATR_SUB     := W_ER * TR_MA12 + (1 - W_ER) * (0.6 * TR_MA24 + 0.4 * TR_MA40);

RAW_BUY_SUB := CROSS(C, DONCHIAN_H) AND (C - DONCHIAN_H) / (ATR_SUB + EPS) > 0.35 AND C > MA20_SUB * MA20_TOL;
KC_UP_SUB   := MA20_SUB + 1.45 * ATR_SUB;
AMT_PRV     := IF(N_BAR > 1, REF(AMOUNT, 1), AMOUNT);
BB_BRK_SUB  := N_BAR >= 20 AND C > MA20_SUB + 2.0 * STD(C, 20) AND AMT_PRED_RAW > AMT_PRV * 0.7;

BUY_T_SUB   := RAW_BUY_SUB OR CROSS(C, KC_UP_SUB) OR BB_BRK_SUB;

OPEN_SAFE   := FROMOPEN > 40 OR NOT(ISLASTBAR);
CORE_SUB    := BUY_T_SUB AND C > MA20_SUB * CORE_TOL;
FLT_SUB     := SZ_VAL > 0.3 OR DARK_FLOW > 0.08 OR CF_FLY_SIGNAL;

FINAL_B_SUB := CORE_SUB AND FLT_SUB AND OPEN_SAFE AND WARM_OK;

SUB_EXIT_COND := CROSS(MA20_SUB, C) OR C < MA20_SUB * EXIT_TOL;
LAST_B      := IF(SUM(FINAL_B_SUB, N_BAR) > 0, BARSLAST(FINAL_B_SUB), 9999);
LAST_X      := IF(SUM(SUB_EXIT_COND, N_BAR) > 0, BARSLAST(SUB_EXIT_COND), 9999);
SUB_HOLD    := LAST_B < LAST_X;

PRICE_HHV10 := C >= IF(N_BAR >= 10, HHV(C, 10), HHV(C, N_BAR));
SZ_HHV10    := SZ_VAL >= IF(N_BAR >= 10, HHV(SZ_VAL, 10), HHV(SZ_VAL, N_BAR));
VOL_WEAK    := AMT_PRED_RAW < AMT_MA_SAFE * 0.75;

SUB_SUMMER := SUB_HOLD AND ((PRICE_HHV10 AND NOT(SZ_HHV10)) OR SZ_VAL > 1.2);
SUB_SPRING := SUB_HOLD AND NOT(SUB_SUMMER);
SUB_WINTER := NOT(SUB_HOLD) AND RAW_EXHAUSTION;
SUB_AUTUMN := NOT(SUB_HOLD) AND NOT(SUB_WINTER);

{==================== 五、四象柱体与动量线渲染 ====================}
{ 少阴·秋 观望区量能柱阴阳精细化区分 }
AUTUMN_UP_BAR := SUB_AUTUMN AND C >= O;
AUTUMN_DN_BAR := SUB_AUTUMN AND C < O;

STICKLINE(AUTUMN_UP_BAR, 0, SZ_VAL, SEASON_BAR_W, 0), COLORB0B0B0;
STICKLINE(AUTUMN_DN_BAR, 0, SZ_VAL, SEASON_BAR_W, 0), COLOR444444;

STICKLINE(SUB_WINTER, 0, SZ_VAL, SEASON_BAR_W, 0), COLORFF00FF;
STICKLINE(SUB_SPRING, 0, SZ_VAL, SEASON_BAR_W, 0), COLOR00FFFF;
STICKLINE(SUB_SUMMER, 0, SZ_VAL, SEASON_BAR_W, 0), COLOR00AAFF;

SZ_CROSS_DN := CROSS(TMC_OBSERVE, SZ_MA_VAL);
STICKLINE(SZ_CROSS_DN, 0, SZ_VAL, 0.6, 0), COLORFF0088;
STICKLINE(SZ_CROSS_DN, 0, SZ_VAL, 0.2, 0), COLORFFFFFF;

STICKLINE(CF_FLY_SIGNAL, 0, SZ_VAL, 2.5, 0), COLOR0088FF;
STICKLINE(CF_FLY_SIGNAL, 0, SZ_VAL, 1.4, 0), COLORCYAN;
STICKLINE(CF_FLY_SIGNAL, 0, SZ_VAL, 0.5, 0), COLORFFFFFF;

SZ_UP_ST    := SZ_MA_VAL >= 0;
SZ_DN_ST    := NOT(SZ_UP_ST);
SZ_UP_PRV   := IF(N_BAR > 1, REF(SZ_UP_ST, 1), SZ_UP_ST);
SZ_DN_PRV   := IF(N_BAR > 1, REF(SZ_DN_ST, 1), SZ_DN_ST);
SZ_MA_LINE  : SZ_MA_VAL, NODRAW;
SZ_MA_UP    : IF(SZ_UP_ST OR SZ_UP_PRV, SZ_MA_VAL, DRAWNULL), COLOR00FFAA, LINETHICK1;
SZ_MA_DN    : IF(SZ_DN_ST OR SZ_DN_PRV, SZ_MA_VAL, DRAWNULL), COLORFF0077, LINETHICK1;

TMC_OBS_LINE: TMC_OBSERVE, COLORFFFFFF, LINETHICK1;
ZERO_LINE   : 0, COLOR777777, DOTLINE;

{==================== 六、四角战术面板 ====================}
IS_LAST      := ISLASTBAR;
ISL_SPRING   := IS_LAST AND SUB_SPRING;

DRAWTEXT_FIX(ISL_SPRING AND NOT(VOL_WEAK), 0.01, 0.02, 0, '气数四季: 少阳·春 (生发主攻)'), COLOR00FFFF;
DRAWTEXT_FIX(ISL_SPRING AND VOL_WEAK,      0.01, 0.02, 0, '气数四季: 少阳·春 (无量虚增)'), COLOR00CCCC;
DRAWTEXT_FIX(IS_LAST AND SUB_SUMMER,       0.01, 0.02, 0, '气数四季: 老阳·夏 (阳极落袋)'), COLOR00AAFF;
DRAWTEXT_FIX(IS_LAST AND SUB_AUTUMN,       0.01, 0.02, 0, '气数四季: 少阴·秋 (覆巢观望)'), COLOR888888;
DRAWTEXT_FIX(IS_LAST AND SUB_WINTER,       0.01, 0.02, 0, '气数四季: 老阴·冬 (暗流潜伏)'), COLORFF00FF;

DRAWTEXT_FIX(IS_LAST, 0.75, 0.02, 0, '暗流控盘:'), COLORFFFFFF;
DRAWNUMBER_FIX(IS_LAST, 0.82, 0.02, 2, DARK_FLOW), COLORYELLOW;

DRAWTEXT_FIX(IS_LAST, 0.01, 0.99, 0, '距起爆点:'), COLORFFFFFF;
DRAWNUMBER_FIX(IS_LAST AND HAS_FLY, 0.09, 0.99, 0, BARS_TO_FLY), COLOR00FFFF;
DRAWTEXT_FIX(IS_LAST AND HAS_FLY, 0.15, 0.99, 0, '根'), COLORFFFFFF;
DRAWTEXT_FIX(IS_LAST AND NOT(HAS_FLY), 0.09, 0.99, 0, '--'), COLOR888888;

DRAWTEXT_FIX(IS_LAST AND SQZ_FIRE, 0.25, 0.99, 0, '【冬至火花·点火】'), COLORFF00FF;
DRAWTEXT_FIX(IS_LAST AND IS_SQUEEZE AND NOT(SQZ_FIRE), 0.23, 0.99, 0, '【极阴挤压·蓄势】'), COLOR00FFFF;
DRAWTEXT_FIX(IS_LAST AND NOT(IS_SQUEEZE), 0.23, 0.99, 0, '【气数平稳】'), COLOR888888;

DRAWTEXT_FIX(IS_LAST, 0.75, 0.99, 0, '多空能量比:'), COLORFFFFFF;
DRAWNUMBER_FIX(IS_LAST, 0.82, 0.99, 2, ENERGY_RATIO), COLORCYAN;
